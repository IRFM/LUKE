\paragraph{\label{fpdiscretemomentum}Fokker-Planck equation}

In this section, only the electron dynamics in momentum space is discussed,
at fixed $\psi _{l+1/2}.$ Hence

\begin{align}
\left. \frac{\widetilde{q}}{B_{0}}p^{2}\frac{\partial f^{\left( 0\right) }}{
\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)} & =\frac{\widetilde {q}%
_{l+1/2}}{B_{0,l+1/2}}\left. \frac{\partial\left( p^{2}\mathbf{S}
_{p}^{\left( 0\right) }\right) }{\partial p}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}  \notag \\
& -\frac{\widetilde{q}_{l+1/2}}{B_{0,l+1/2}}\frac{p_{i+1/2}}{\lambda
^{l+1/2,j+1/2}}\times  \notag \\
& \left. \frac{\partial}{\partial\xi_{0}}\left( \sqrt{1-\xi_{0}^{2}}\lambda%
\mathbf{S}_{\xi}^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{align}
\label{discretemomentflux} where 
\begin{equation}
\left\{ 
\begin{array}{l}
S_{p}^{\left( 0\right) }=-D_{pp}^{\left( 0\right) }\frac{\partial
f_{0}^{\left( 0\right) }}{\partial p}+D_{p\xi}^{\left( 0\right) }\frac{\sqrt{%
1-\xi_{0}^{2}}}{p}\frac{\partial f_{0}^{\left( 0\right) }}{\partial\xi_{0}}%
+F_{p}^{\left( 0\right) }f_{0}^{\left( 0\right) } \\ 
S_{\xi}^{\left( 0\right) }=-D_{\xi p}^{\left( 0\right) }\frac{\partial
f_{0}^{\left( 0\right) }}{\partial p}+D_{\xi\xi}^{\left( 0\right) }\frac{%
\sqrt{1-\xi_{0}^{2}}}{p}\frac{\partial f_{0}^{\left( 0\right) }}{%
\partial\xi_{0}}+F_{\xi}^{\left( 0\right) }f_{0}^{\left( 0\right) }%
\end{array}
\right.
\end{equation}

Here, diffusion cross-terms between momentum and configuration spaces are
neglected, i.e. $D_{p\psi }^{\left( 0\right) }=D_{\psi p}^{\left( 0\right)
}=D_{\xi \psi }^{\left( 0\right) }=D_{\psi \xi }^{\left( 0\right) }=0.$ By
definition, the discretization procedure must preserve the conservative
nature of the momentum dynamics, as discussed in Sec. \ref{fluxcons}, and
therefore, careful attention must be paid how to proceed from Eq. \ref%
{discretemomentflux}, in particular when diffusion or convection
coefficients vary strongly on a grid step. Moreover, boundary conditions on
flux grid must be as much as possible naturaly satisfied by the discrete
form of the Fokker-Planck equation. Indeed, cross-derivatives never satisfy
this condition simultaneously along the momentum and pitch-angle directions,
and consequently additional external boundary conditions must be added to
enforce this condition.

There are several ways to perform the discretization of the
cross-derivatives. Since $D_{p\xi }^{\left( 0\right) }$ and $D_{\xi
p}^{\left( 0\right) }$ terms do not contribute to the Maxwellian solution
when no external perturbation is applied, it is not necessary to perform an
interpolation between full and half grids for cross-derivatives in order to
ensure a correct numerical flux balance in momentum space. In that case, the
usual procedure is simply to center differences, as done in this section.
However, there is a much more complex approach, but also more consistent one
with respect to the use two grids, one for the fluxes and the other for the
distribution, which is also presented in Appendix \ref%
{crossderivativesappendix}. However, because of its complexity, it has not
been yet implemented in the code even if this approach is in principle
better designed for non-uniform grids with strong localized variations of $%
D_{p\xi }^{\left( 0\right) }$ and $D_{\xi p}^{\left( 0\right) }.$

The starting point of the discretization procedure is the following
expressions 
\begin{align}
\frac{\partial \left( p^{2}S_{p}^{\left( 0\right) }\right) }{\partial p}& =%
\frac{\partial }{\partial p}\left( -p^{2}D_{pp}^{\left( 0\right) }\frac{%
\partial f_{0}^{\left( 0\right) }}{\partial p}+p^{2}F_{p}^{\left( 0\right)
}f_{0}^{\left( 0\right) }\right)  \notag \\
& +\sqrt{1-\xi _{0}^{2}}\frac{\partial }{\partial p}\left( pD_{p\xi
}^{\left( 0\right) }\right) \frac{\partial f_{0}^{\left( 0\right) }}{%
\partial \xi _{0}}  \notag \\
& +\sqrt{1-\xi _{0}^{2}}pD_{p\xi }^{\left( 0\right) }\frac{\partial
^{2}f_{0}^{\left( 0\right) }}{\partial p\partial \xi _{0}}
\end{align}
\begin{align}
\frac{\partial \left( \sqrt{1-\xi _{0}^{2}}\lambda S_{\xi }^{\left( 0\right)
}\right) }{\partial \xi _{0}}& =\frac{\partial }{\partial \xi _{0}}\left(
D_{\xi \xi }^{\left( 0\right) }\frac{1-\xi _{0}^{2}}{p}\lambda \frac{%
\partial f_{0}^{\left( 0\right) }}{\partial \xi _{0}}\right.  \notag \\
& \left. +\sqrt{1-\xi _{0}^{2}}\lambda F_{\xi }^{\left( 0\right)
}f_{0}^{\left( 0\right) }\right)  \notag \\
& -\frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}\lambda
D_{\xi p}^{\left( 0\right) }\right) \frac{\partial f_{0}^{\left( 0\right) }}{%
\partial p}  \notag \\
& -\sqrt{1-\xi _{0}^{2}}\lambda D_{\xi p}^{\left( 0\right) }\frac{\partial
^{2}f_{0}^{\left( 0\right) }}{\partial \xi _{0}\partial p}
\end{align}
where cross-derivatives terms have been developped directly from analytical
formulas, and considered in separately. Here, it is assumed in an implicit
manner that derivatives of $pD_{p\xi }^{\left( 0\right) }$ and $\sqrt{1-\xi
_{0}^{2}}\lambda D_{\xi p}^{\left( 0\right) }$ exists.

Hence 
\begin{equation}
\left. \frac{\widetilde{q}}{B_{0}}p^{2}\frac{\partial f_{0}^{\left( 0\right)
}}{\partial t}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{\widetilde{q}
_{l+1/2}}{B_{0,l+1/2}}\sum_{m=1}^{m=8}T^{\left[ m\right] }
\end{equation}
with 
\begin{equation}
T^{\left[ 1\right] }=\frac{-p_{i+1}^{2}D_{pp,l+1/2,i+1,j+1/2}^{\left(
0\right) }\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1,j+1/2}^{(k+1)}+p_{i+1}^{2}F_{p,l+1/2,i+1,j+1/2}^{\left( 0\right)
}f_{0,l+1/2,i+1,j+1/2}^{\left( 0\right) (k+1)}}{\Delta p_{i+1/2}}
\end{equation}
\begin{equation}
T^{\left[ 2\right] }=\frac{p_{i}^{2}D_{pp,l+1/2,i,j+1/2}^{\left( 0\right)
}\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i,j+1/2}^{(k+1)}-p_{i}^{2}F_{p,l+1/2,i,j+1/2}^{\left( 0\right)
}f_{0,l+1/2,i,j+1/2}^{\left( 0\right) (k+1)}}{\Delta p_{i+1/2}}
\end{equation}
\begin{equation}
T^{\left[ 3\right] }=+\sqrt{1-\xi _{0,j+1/2}^{2}}\left. \frac{\partial
\left( pD_{p\xi }^{\left( 0\right) }\right) }{\partial p}\right|
_{l+1/2,i+1/2,j+1/2}\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial
\xi }\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}
\begin{equation}
T^{\left[ 4\right] }=+\sqrt{1-\xi _{0,j+1/2}^{2}}p_{i+1/2}D_{p\xi
,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }\left. \frac{\partial
^{2}f_{0}^{\left( 0\right) }}{\partial p\partial \xi }\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}
\begin{align}
T^{\left[ 5\right] }& =-\frac{p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}\left[ \frac{%
D_{\xi \xi ,l+1/2,i+1/2,j+1}^{\left( 0\right) }\left( 1-\xi
_{0,j+1}^{2}\right) \lambda ^{l+1/2,j+1}\left. \frac{\partial f_{0}^{\left(
0\right) }}{\partial \xi }\right| _{l+1/2,i+1/2,j+1}^{(k+1)}}{%
p_{i+1/2}\Delta \xi _{j+1/2}}\right.  \notag \\
& +\left. \frac{p_{i+1/2}\sqrt{1-\xi _{0,j+1}^{2}}\lambda ^{l+1/2,j+1}F_{\xi
,l+1/2,i+1/2,j+1}^{\left( 0\right) }f_{0,l+1/2,i+1/2,j+1}^{\left( 0\right)
(k+1)}}{\Delta \xi _{j+1/2}}\right]
\end{align}
\begin{align}
T^{\left[ 6\right] }& =\frac{p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}\left[ \frac{
D_{\xi \xi ,l+1/2,i+1/2,j}^{\left( 0\right) }\left( 1-\xi _{0,j}^{2}\right)
\lambda ^{l+1/2,j}\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial
\xi }\right| _{l+1/2,i+1/2,j}^{(k+1)}}{p_{i+1/2}\Delta \xi _{j+1/2}}\right. 
\notag \\
& +\left. \frac{p_{i+1/2}\sqrt{1-\xi _{0,j}^{2}}\lambda ^{l+1/2,j}F_{\xi
,l+1/2,i+1/2,j}^{\left( 0\right) }f_{0,l+1/2,i+1/2,j}^{\left( 0\right) (k+1)}%
}{\Delta \xi _{j+1/2}}\right]
\end{align}

\begin{equation}
T^{\left[ 7\right] }=\frac{p_{i+1/2}}{\lambda^{l+1/2,j+1/2}}\left. \frac{%
\partial}{\partial\xi_{0}}\left( \sqrt{1-\xi_{0}^{2}}\lambda D_{\xi
p}^{\left( 0\right) }\right) \right| _{l+1/2,i+1/2,j+1/2}\left. \frac{%
\partial f_{0}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}
\begin{equation}
T^{\left[ 8\right] }=\frac{p_{i+1/2}}{\lambda^{l+1/2,j+1/2}}\sqrt {%
1-\xi_{0,j+1/2}^{2}}\lambda^{l+1/2,j+1/2}D_{\xi p,l+1/2,i+1/2,j+1/2}^{\left(
0\right) }\left. \frac{\partial^{2}f_{0}^{\left( 0\right) }}{\partial
\xi_{0}\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}

Discrete expressions of the partial derivatives are, 
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1,j+1/2}^{(k+1)}=\frac{f_{0,l+1/2,i+3/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta p_{i+1}}
\end{equation}
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{f_{0,l+1/2,i+3/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i-1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta
p_{i+1}+\Delta p_{i}}  \label{dfdp_ij}
\end{equation}
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i,j+1/2}^{(k+1)}=\frac{f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i-1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta p_{i}}
\label{dfdp_imj}
\end{equation}
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial\xi_{0}}\right|
_{l+1/2,i+1/2,j+1}^{(k+1)}=\frac{f_{0,l+1/2,i+1/2,j+3/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta\xi_{0,j+1}}
\end{equation}
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial\xi_{0}}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{f_{0,l+1/2,i+1/2,j+3/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i+1/2,j-1/2}^{\left( 0\right) (k+1)}}{\Delta\xi
_{0,j+1}+\Delta\xi_{0,j}}  \label{dfdxi_ij}
\end{equation}
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial\xi_{0}}\right|
_{l+1/2,i+1/2,j}^{(k+1)}=\frac{f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i+1/2,j-1/2}^{\left( 0\right) (k+1)}}{\Delta\xi_{0,j}}
\label{dfdxi_ijm}
\end{equation}
and cross-derivatives 
\begin{align}
\left. \frac{\partial^{2}f_{0}^{\left( 0\right) }}{\partial p\partial \xi_{0}%
}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)} & =\left. \frac{\partial
^{2}f_{0}^{\left( 0\right) }}{\partial\xi_{0}\partial p}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}  \notag \\
& =\frac{f_{0,l+1/2,i+3/2,j+3/2}^{\left( 0\right)
(k+1)}+f_{0,l+1/2,i-1/2,j-1/2}^{\left( 0\right) (k+1)}}{\left( \Delta
p_{i+1}+\Delta p_{i}\right) \left( \Delta\xi_{0,j+1}+\Delta\xi_{0,j}\right) }
\notag \\
& -\frac{f_{0,l+1/2,i+3/2,j-1/2}^{\left( 0\right)
(k+1)}+f_{0,l+1/2,i-1/2,j+3/2}^{\left( 0\right) (k+1)}}{\left( \Delta
p_{i+1}+\Delta p_{i}\right) \left( \Delta\xi_{0,j+1}+\Delta\xi_{0,j}\right) }
\end{align}
while other derivatives become in discrete form 
\begin{equation}
\left. \frac{\partial\left( pD_{p\xi}^{\left( 0\right) }\right) }{\partial p}%
\right| _{l+1/2,i+1/2,j+1/2}=\frac{p_{i+1}D_{p\xi ,l+1/2,i+1,j+1/2}^{\left(
0\right) }-p_{i}D_{p\xi,l+1/2,i,j+1/2}^{\left( 0\right) }}{\Delta p_{i+1/2}}
\end{equation}
and 
\begin{align}
& \left. \frac{\partial}{\partial\xi_{0}}\left( \sqrt{1-\xi_{0}^{2}}\lambda
D_{\xi p}^{\left( 0\right) }\right) \right| _{l+1/2,i+1/2,j+1/2}  \notag \\
& =\frac{\sqrt{1-\xi_{0,j+1}^{2}}\lambda^{l+1/2,j+1}D_{\xi
p,l+1/2,i+1/2,j+1}^{\left( 0\right) }}{\Delta\xi_{0,j+1/2}}  \notag \\
& -\frac{\sqrt{1-\xi_{0,j}^{2}}\lambda^{l+1/2,j}D_{\xi
p,l+1/2,i+1/2,j}^{\left( 0\right) }}{\Delta\xi_{0,j+1/2}}
\end{align}

By definition, cross-derivatives are symmetric. Furthermore, the derivative 
\begin{equation}
\frac{\partial}{\partial\xi_{0}}\left( \sqrt{1-\xi_{0}^{2}}\lambda D_{\xi
p}^{\left( 0\right) }\right)
\end{equation}
must be taken on the flux grid in order to fullfil naturally boundary
condition at $\left| \xi_{0}\right| =1$ and avoid to specify $D_{\xi
p}^{\left( 0\right) }$ at this momentum space location. It is a crucial
point, especially for the Lower Hybrid and the Ohmic current drive problems,
since this domain of the momentum space plays a very important role for the
determination of the plasma current level.

Since the distribution function is defined on the half grid, while fluxes on
the full grid, it is necessary to interpolate, because in some derivatives,
values of $f_{0}^{\left( 0\right) }$ are taken on the full grid. In a
general way, whatever the detailed value of the weighting factor $%
\delta^{\left( 0\right) }$ which will be discussed in the Sec. \ref%
{fpgridinterpolation}, one may write for the terms proportional to $%
D_{pp}^{\left( 0\right) }$ and $F_{p}^{\left( 0\right) },$%
\begin{align}
f_{0,l+1/2,i+1,j+1/2}^{\left( 0\right) (k+1)} & =\left( 1-\delta
_{p,l+1/2,i+1,j+1/2}^{\left( 0\right) }\right)
f_{0,l+1/2,i+3/2,j+1/2}^{\left( 0\right) (k+1)}  \notag \\
& +\delta_{p,l+1/2,i+1,j+1/2}^{\left( 0\right)
}f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{align}
\begin{align}
f_{0,l+1/2,i,j+1/2}^{\left( 0\right) (k+1)} & =\left( 1-\delta
_{p,l+1/2,i,j+1/2}^{\left( 0\right) }\right) f_{0,l+1/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}  \notag \\
& +\delta_{p,l+1/2,i,j+1/2}^{\left( 0\right)
}f_{0,l+1/2,i-1/2,j+1/2}^{\left( 0\right) (k+1)}  \label{f_imj}
\end{align}
and for terms proportional to $D_{\xi\xi}^{\left( 0\right) }$ and $F_{\xi
}^{\left( 0\right) }$%
\begin{align}
f_{0,i+1/2,j+1}^{\left( 0\right) (k+1)} & =\left( 1-\delta_{\xi
,l+1/2,i+1/2,j+1}^{\left( 0\right) }\right) f_{0,l+1/2,i+1/2,j+3/2}^{\left(
0\right) (k+1)}  \notag \\
& +\delta_{\xi,l+1/2,i+1/2,j+1}^{\left( 0\right)
}f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{align}
\begin{align}
f_{0,i+1/2,j}^{\left( 0\right) (k+1)} & =\left( 1-\delta_{\xi
,l+1/2,i+1/2,j}^{\left( 0\right) }\right) f_{0,l+1/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}  \label{f_ijm} \\
& +\delta_{\xi,l+1/2,i+1/2,j}^{\left( 0\right)
}f_{0,l+1/2,i+1/2,j-1/2}^{\left( 0\right) (k+1)}
\end{align}

Gathering all terms, corresponding matrix coefficients for the zero order
Fokker-Planck equation may be expressed as 
\begin{equation}
\left. \frac{\widetilde{q}}{B_{0}}p^{2}\frac{\partial f^{\left( 0\right) }}{%
\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{\widetilde{q}_{l+1/2}}{
B_{0,l+1/2}}\sum_{i^{\prime}=i-1}^{i^{\prime}=i+1}\sum_{j^{\prime
}=j-1}^{j^{\prime}=j+1}\overline{\overline{M}}_{p,l+1/2,i^{\prime
}+1/2,j^{\prime}+1/2}^{\left( 0\right) }f_{0,l+1/2,i^{\prime}+1/2,j^{\prime
}+1/2}^{\left( 0\right) (k+1)}
\end{equation}
where 
\begin{align}
\overline{\overline{M}}_{p,l+1/2,i+3/2,j+3/2}^{\left( 0\right) } & =\frac{
p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi_{0,j+1/2}^{2}}}{%
\Delta\xi_{0,j+1}+\Delta\xi_{0,j}}D_{p\xi,l+1/2,i+1/2,j+1/2}^{\left(
0\right) }  \notag \\
& +\frac{p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi
_{0,j+1/2}^{2}}}{\Delta\xi_{0,j+1}+\Delta\xi_{0,j}}D_{\xi
p,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }
\end{align}
\begin{align}
\overline{\overline{M}}_{p,l+1/2,i+1/2,j+3/2}^{\left( 0\right) } & =\frac{
p_{i+1}}{\Delta p_{i+1/2}}\frac{\sqrt{1-\xi_{0,j+1/2}^{2}}}{\Delta
\xi_{0,j+1}+\Delta\xi_{0,j}}D_{p\xi,l+1/2,i+1,j+1/2}^{\left( 0\right) } 
\notag \\
& -\frac{p_{i}}{\Delta p_{i+1/2}}\frac{\sqrt{1-\xi_{0,j+1/2}^{2}}}{\Delta
\xi_{0,j+1}+\Delta\xi_{0,j}}D_{p\xi,l+1/2,i,j+1/2}^{\left( 0\right) }  \notag
\\
& -\frac{1-\xi_{0,j+1}^{2}}{\Delta\xi_{0,j+1/2}\Delta\xi_{0,j+1}}\frac{%
\lambda^{l+1/2,j+1}}{\lambda^{l+1/2,j+1/2}}D_{\xi\xi,l+1/2,i+1/2,j+1}^{%
\left( 0\right) }  \notag \\
& -p_{i+1/2}\frac{\sqrt{1-\xi_{0,j+1}^{2}}}{\Delta\xi_{0,j+1/2}}\left(
1-\delta_{\xi,l+1/2,i+1/2,j+1}^{\left( 0\right) }\right) \times  \notag \\
& \frac{\lambda^{l+1/2,j+1}}{\lambda^{l+1/2,j+1/2}}F_{\xi,l+1/2,i+1/2,j+1}^{%
\left( 0\right) }
\end{align}
\begin{align}
\overline{\overline{M}}_{p,l+1/2,i-1/2,j+3/2}^{\left( 0\right) } & =-\frac{
p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi_{0,j+1/2}^{2}}}{%
\Delta\xi_{0,j+1}+\Delta\xi_{0,j}}D_{p\xi,l+1/2,i+1/2,j+1/2}^{\left(
0\right) }  \notag \\
& -\frac{p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi
_{0,j+1/2}^{2}}}{\Delta\xi_{0,j+1}+\Delta\xi_{0,j}}D_{\xi
p,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }
\end{align}
\begin{align}
\overline{\overline{M}}_{p,l+1/2,i+3/2,j+1/2}^{\left( 0\right) } & =-\frac{%
p_{i+1}^{2}}{\Delta p_{i+1}\Delta p_{i+1/2}}D_{pp,l+1/2,i+1,j+1/2}^{\left(
0\right) }  \notag \\
& +\frac{p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi_{0,j+1}^{2}%
}}{\Delta\xi_{0,j+1/2}}\times  \notag \\
& \frac{\lambda^{l+1/2,j+1}}{\lambda^{l+1/2,j+1/2}}D_{\xi
p,l+1/2,i+1/2,j+1}^{\left( 0\right) }  \notag \\
& -\frac{p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi_{0,j}^{2}}%
}{\Delta\xi_{0,j+1/2}}\times  \notag \\
& \frac{\lambda^{l+1/2,j}}{\lambda^{l+1/2,j+1/2}}D_{\xi
p,l+1/2,i+1/2,j}^{\left( 0\right) }  \notag \\
& +\frac{p_{i+1}^{2}}{\Delta p_{i+1/2}}\left(
1-\delta_{p,l+1/2,i+1,j+1/2}^{\left( 0\right) }\right)
F_{p,l+1/2,i+1,j+1/2}^{\left( 0\right) }
\end{align}
\begin{align}
\overline{\overline{M}}_{p,i+1/2,j+1/2}^{\left( 0\right) } & =\frac{%
p_{i+1}^{2}}{\Delta p_{i+1/2}\Delta p_{i+1}}D_{pp,l+1/2,i+1,j+1/2}^{\left(

\paragraph{Knock-on source term}

The knock-on source term must be evaluated at each radial position $\psi
_{l+1/2},$ and also at each momentum and pitch-angle positions $\left(
p_{i+1/2},\xi _{0,j+1/2}\right) $in momentum space. However, the poloidal
position of the birth of the secondary electron plays a crucial role in the
calculations, that significantly increases the difficulty of the problem.

As discussed in Sec. \ref{runawaysection},

\begin{equation}
\mathcal{S}_{R}^{\left( 0\right) }=\frac{\int_{-\infty }^{\overline{t}%
}\left\langle \Gamma _{R}\right\rangle _{V}d\overline{t}}{\pi \ln \Lambda
^{\dag }}\frac{1}{p}\frac{1}{\gamma \left( \gamma -1\right) }\frac{B_{0}}{%
\lambda \widetilde{q}}\frac{\left| \xi _{0}\right| }{\left( 1-\xi
_{0}^{2}\right) ^{2}}\left[ \frac{1}{\left| \widehat{\psi }\cdot \widehat{r}%
\right| }\frac{r}{B_{P}}\frac{1}{\left| \Psi ^{^{\prime }}\right| }\right]
_{\theta ^{*}}
\end{equation}
where $\gamma =\sqrt{1+\beta _{th}^{\dag 2}p^{2}}$, $\xi ^{*2}=\sqrt{\left(
\gamma -1\right) /\left( \gamma +1\right) }$ and the poloidal angle $\theta
^{*}$ is defined by the relation 
\begin{equation}
\Psi \left( \psi ,\theta ^{*}\right) =\frac{1-\xi ^{*2}}{1-\xi _{0}^{2}}=%
\frac{2}{\left( 1+\gamma \right) \left( 1-\xi _{0}^{2}\right) }
\end{equation}
$.$On the discrete mesh, the source term $\mathcal{S}_{R}^{\left( 0\right) } 
$ at grid point $\left( l+1/2,i+1/2,j+1/2\right) $ becomes 
\begin{eqnarray}
\mathcal{S}_{R,l+1/2,i+1/2,j+1/2}^{\left( 0\right) } &=&\frac{\left.
\int_{-\infty }^{\overline{t}}\left\langle \Gamma _{R}\right\rangle _{V}d%
\overline{t}\right| _{l+1/2}}{\pi \ln \Lambda ^{\dag }R_{p}}\frac{1}{%
p_{i+1/2}}\frac{1}{\gamma _{i+1/2}\left( \gamma _{i+1/2}-1\right) }\frac{%

_{0,j+1/2}\right| }{\left( 1-\xi _{0,j+1/2}^{2}\right) ^{2}}  \notag \\
&&\times \left[ \frac{r}{\left| \widehat{\psi }\cdot \widehat{r}\right| }%
\frac{1}{B_{P}}\frac{1}{\left| \Psi ^{^{\prime }}\right| }\right] _{\theta
_{l+1/2,i+1/2,j+1/2}^{*}}
\end{eqnarray}
where $\theta _{l+1/2,i+1/2,j+1/2}^{*}$ is determined from the relation 
\begin{equation}
\Psi \left( \psi _{l+1/2},\theta _{l+1/2,i+1/2,j+1/2}^{*}\right) =\frac{2}{%
\left( 1+\gamma _{i+1/2}\right) \left( 1-\xi _{0,j+1/2}^{2}\right) }

on each flux surface $\psi _{l+1/2}.$ Here 
\begin{equation}
\left| \Psi ^{^{\prime }}\right| _{\theta _{l+1/2,i+1/2,j+1/2}^{*}}=\frac{1}{%
B_{0,l+1/2}}\frac{dB}{d\theta }\left( \psi _{l+1/2},\theta

the integration domain, the source term $\mathcal{S}_{R,l+1/2,i+1/2,j+1/2}^{%
