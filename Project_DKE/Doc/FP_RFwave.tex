\smallskip In the Fokker-Planck equation, the diffusion and convection
elements are bounce-averaged according to the expressions 
(\ref{diffbavpp}) and
(\ref{convbavm}), which gives, using (\ref{dtensor1}-\ref{dtensor4})
\begin{align}
D_{pp}^{\text{RF}(0)}  &  =\left\{  \sum_{n=-\infty}^{+\infty}\sum_{b}%
(1-\xi^{2})D_{b,n}^{\text{RF}}(p,\xi)\right\} \\
D_{p\xi}^{\text{RF}(0)}  &  =\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}%
\xi_{0}}\sum_{n=-\infty}^{+\infty}\sum_{b}-\frac{\sqrt{1-\xi^{2}}}{\xi}\left[
1-\xi^{2}-\frac{n\Omega}{\omega_{b}}\right]  D_{b,n}^{\text{RF}}%
(p,\xi)\right\} \nonumber\\
D_{\xi p}^{\text{RF}(0)}  &  =\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}%
\xi_{0}}\sum_{n=-\infty}^{+\infty}\sum_{b}-\frac{\sqrt{1-\xi^{2}}}{\xi}\left[
1-\xi^{2}-\frac{n\Omega}{\omega_{b}}\right]  D_{b,n}^{\text{RF}}%
(p,\xi)\right\} \nonumber\\
D_{\xi\xi}^{\text{RF}(0)}  &  =\left\{  \frac{\xi^{2}}{\Psi\xi_{0}^{2}}%
\sum_{n=-\infty}^{+\infty}\sum_{b}\frac{1}{\xi^{2}}\left[  1-\xi^{2}%
-\frac{n\Omega}{\omega_{b}}\right]  ^{2}D_{b,n}^{\text{RF}}(p,\xi)\right\}
\nonumber
\end{align}
where ()
\begin{equation}
\left\{  \mathcal{O}\right\}  =\frac{1}{\lambda\widetilde{q}}\left[  \frac
{1}{2}\sum\limits_{\sigma}\right]  _{T}\int_{\theta_{\min}}^{\theta_{\max}%
}\frac{d\theta}{2\pi}\frac{1}{\left|  \widehat{\psi}\cdot\widehat{r}\right|
}\frac{r}{R_{p}}\frac{B}{B_{P}}\frac{\xi_{0}}{\xi}\mathcal{O}%
\end{equation}

Since $1-\xi^{2}=\Psi\left(  1-\xi_{0}^{2}\right)  $,
\begin{align}
D_{pp}^{\text{RF}(0)}  &  =\sum_{n=-\infty}^{+\infty}\sum_{b}(1-\xi_{0}%
^{2})D_{b,n}^{\text{RF}(0)}(p,\xi_{0})\\
D_{p\xi}^{\text{RF}(0)}  &  =\sum_{n=-\infty}^{+\infty}\sum_{b}-\frac
{\sqrt{1-\xi_{0}^{2}}}{\xi_{0}}\left[  1-\xi_{0}^{2}-\frac{n\Omega_{0}}%
{\omega_{b}}\right]  D_{b,n}^{\text{RF}(0)}(p,\xi_{0})\nonumber\\
D_{\xi p}^{\text{RF}(0)}  &  =\sum_{n=-\infty}^{+\infty}\sum_{b}-\frac
{\sqrt{1-\xi_{0}^{2}}}{\xi_{0}}\left[  1-\xi_{0}^{2}-\frac{n\Omega_{0}}%
{\omega_{b}}\right]  D_{b,n}^{\text{RF}(0)}(p,\xi_{0})\nonumber\\
D_{\xi\xi}^{\text{RF}(0)}  &  =\sum_{n=-\infty}^{+\infty}\sum_{b}\frac{1}%
{\xi_{0}^{2}}\left[  1-\xi_{0}^{2}-\frac{n\Omega_{0}}{\omega_{b}}\right]
^{2}D_{b,n}^{\text{RF}(0)}(p,\xi_{0})\nonumber
\end{align}
where
\begin{equation}
D_{b,n}^{\text{RF}(0)}(p,\xi_{0})=\left\{  \Psi D_{b,n}^{\text{RF}}%
(p,\xi)\right\}  \label{bouncedrf_fp}%
\end{equation}
and $\Omega_{0}$ is the cyclotron frequency taken at the minimum $B$ value,
according to the relation
\begin{equation}
\Omega_{0}=\frac{eB_{0}}{\gamma m_{e}}=-\frac{\omega_{ce,0}}{\gamma}%
=\frac{\Omega}{\Psi} \label{omega0}%
\end{equation}
with
\begin{equation}
\omega_{ce,0}=\frac{eB_{0}}{m_{e}}%
\end{equation}

Therefore, only the bounce averaged diffusion coefficient $D_{b,n}%
^{\text{RF}(0)}(p,\xi_{0})$ has to be calculated for the different types of RF
waves, keeping the same formalism. It is determined by inserting expression
(\ref{difcoeffnorm2}) into (\ref{bouncedrf_fp}),%

\begin{equation}
\overline{D}_{b,n}^{\text{RF}(0)}(p,\xi_{0})=\left\{  \Psi\overline{D}%
_{b,n,0}^{\text{RF}}\left|  \Theta_{\mathbf{k}}^{b,\left(  n\right)  }\right|
^{2}\frac{\gamma p_{Te}}{p\left|  \xi\right|  }\delta\left(  N_{\Vert
b}-N_{\Vert\text{res}}\right)  2\pi\delta\left(  \theta-\theta_{b}\right)
\right\}
\end{equation}
which gives
\begin{equation}
\overline{D}_{b,n}^{\text{RF}(0)}(p,\xi_{0})=\frac{\gamma p_{Te}}{p\left|
\xi_{0}\right|  }\left\{  \frac{\xi_{0}}{\xi}\Psi\overline{D}_{b,n,0}%
^{\text{RF}}\left|  \Theta_{\mathbf{k}}^{b,\left(  n\right)  }\right|
^{2}2\pi\delta\left(  \theta-\theta_{b}\right)  \delta\left(  N_{b\Vert
}-N_{\Vert\text{res}}\right)  \right\}
\end{equation}

We perform the poloidal integration
\begin{align}
&  \left\{  \frac{\xi_{0}}{\xi}\Psi\overline{D}_{b,n,0}^{\text{RF}}\left|
\Theta_{\mathbf{k}}^{b,\left(  n\right)  }\right|  ^{2}2\pi\delta\left(
\theta-\theta_{b}\right)  \delta\left(  N_{b\Vert}-N_{\Vert\text{res}}\right)
\right\}  \nonumber\\
&  =\frac{1}{\lambda\widetilde{q}}\left[  \frac{1}{2}\sum\limits_{\sigma
}\right]  _{T}\int_{\theta_{\min}}^{\theta_{\max}}d\theta\frac{1}{\left|
\widehat{\psi}\cdot\widehat{r}\right|  }\frac{r}{R_{p}}\frac{B}{B_{P}}%
\frac{\xi_{0}^{2}}{\xi^{2}}\Psi\overline{D}_{b,n,0}^{\text{RF}}\left|
\Theta_{\mathbf{k}}^{b,\left(  n\right)  }\right|  ^{2}\delta\left(
\theta-\theta_{b}\right)  \delta\left(  N_{b\Vert}-N_{\Vert\text{res}}\right)
\\
&  =\frac{1}{\lambda\widetilde{q}}\frac{r_{\theta_{b}}}{R_{p}}\frac
{B^{\theta_{b}}}{B_{P}^{\theta_{b}}}\frac{\xi_{0}^{2}}{\xi_{\theta_{b}}^{2}%
}\Psi_{\theta_{b}}\overline{D}_{b,n,0}^{\text{RF,}\theta_{b}}H\left(
\theta_{b}-\theta_{\min}\right)  H\left(  \theta_{\max}-\theta_{b}\right)
\left[  \frac{1}{2}\sum\limits_{\sigma}\right]  _{T}\delta\left(  N_{b\Vert
}-N_{\Vert\text{res}}^{\theta_{b}}\right)  \left|  \Theta_{\mathbf{k,}%
\theta_{b}}^{b,\left(  n\right)  }\right|  ^{2}%
\end{align}
with%
\begin{align}
\widehat{\psi}_{\theta_{b}} &  =\widehat{\psi}(\psi,\theta_{b})\\
r_{\theta_{b}} &  =r\left(  \psi,\theta_{b}\right)  \\
R_{\theta_{b}} &  =R\left(  \psi,\theta_{b}\right)  \\
B^{\theta_{b}} &  =B\left(  \psi,\theta_{b}\right)  \\
B_{P}^{\theta_{b}} &  =B_{P}\left(  \psi,\theta_{b}\right)  \\
\Psi_{\theta_{b}} &  =\frac{B\left(  \psi,\theta_{b}\right)  }{B_{0}\left(
\psi\right)  }=\frac{B^{\theta_{b}}}{B_{0}}\\
\xi_{\theta_{b}} &  =\xi\left(  \psi,\theta_{b},\xi_{0}\right)  =\sigma
\sqrt{1-\Psi_{\theta_{b}}\left(  1-\xi_{0}^{2}\right)  }%
\end{align}
and where $H\left(  x\right)  $ is the usual Heaviside step function

Therefore,
\begin{align}
\overline{D}_{b,n}^{\text{RF}(0)}(p,\xi_{0}) &  =\frac{\gamma p_{Te}}{p\left|
\xi_{0}\right|  }\frac{1}{\lambda\widetilde{q}}\frac{r_{\theta_{b}}}{R_{p}%
}\frac{B^{\theta_{b}}}{B_{P}^{\theta_{b}}}\frac{\xi_{0}^{2}}{\xi_{\theta_{b}%
}^{2}}\Psi_{\theta_{b}}\overline{D}_{b,n,0}^{\text{RF,}\theta_{b}}%
\times\nonumber\\
&  H\left(  \theta_{b}-\theta_{\min}\right)  H\left(  \theta_{\max}-\theta
_{b}\right)  \left[  \frac{1}{2}\sum\limits_{\sigma}\right]  _{T}\delta\left(
N_{b\Vert}-N_{\Vert\text{res}}^{\theta_{b}}\right)  \left|  \Theta
_{\mathbf{k,}\theta_{b}}^{b,\left(  n\right)  }\right|  ^{2}\label{dbnb}%
\end{align}
where we define, using (\ref{thetakn2}), (\ref{nparres2}) and (\ref{drf0}),%
\begin{equation}
\Theta_{\mathbf{k,}\theta_{b}}^{b,\left(  n\right)  }=\frac{1}{\sqrt{2}%
}e_{b0,+}e^{-i\alpha_{b}}J_{n-1}\left(  z_{b}^{\theta_{b}}\right)  +\frac
{1}{\sqrt{2}}e_{b0,-}e^{+i\alpha_{b}}J_{n+1}\left(  z_{b}^{\theta_{b}}\right)
+\frac{\xi_{\theta_{b}}}{\sqrt{\Psi_{\theta_{b}}\left(  1-\xi_{0}^{2}\right)
}}e_{b0,\Vert}J_{n}\left(  z_{b}^{\theta_{b}}\right)  \label{thetakn3}%
\end{equation}%
\begin{equation}
N_{\Vert\text{res}}^{\theta_{b}}=\frac{1}{\beta_{Te}}\frac{p_{Te}}%
{p\xi_{\theta_{b}}}\left(  \gamma-\frac{n^{\prime}\Psi_{\theta_{b}}%
\omega_{ce,0}}{\omega_{b}}\right)
\end{equation}%
\begin{equation}
\overline{D}_{b,n,0}^{\text{RF,}\theta_{b}}=\frac{1}{r_{\theta_{b}}%
R_{\theta_{b}}}\frac{1}{m_{e}\ln\Lambda}\frac{1}{\omega_{b}\omega_{pe}^{2}%
}\frac{f_{\text{inc},b}^{l+1/2}}{\left|  \mathbf{\Phi}_{b}\right|
}P_{b,\text{inc}}%
\end{equation}
with
\begin{align}
z_{b}^{\theta_{b}} &  =\frac{k_{b\bot}v_{\bot}^{\theta_{b}}}{\Omega
_{\theta_{b}}}\nonumber\\
&  =-N_{b\bot}\frac{\omega_{b}}{\omega_{ce}^{\theta_{b}}}\frac{p}{m_{e}c}%
\sqrt{1-\xi_{\theta_{b}}^{2}}\nonumber\\
&  =-N_{b\bot}\frac{\omega_{b}}{\omega_{ce,0}}\frac{p}{m_{e}c}\frac
{\sqrt{1-\xi_{0}^{2}}}{\sqrt{\Psi_{\theta_{b}}}}\label{zb2}%
\end{align}
