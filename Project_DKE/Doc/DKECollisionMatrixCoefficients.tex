A partir des composantes diffusives et convectives de l'operateur
de collisions,

\[
\left\{ \begin{array}{l}
\widetilde{D}_{pp}^{\text{C}}=A(p)\\
\widetilde{D}_{p\xi}^{\text{C}}=0\\
\widetilde{D}_{\xi p}^{\text{C}}=0\\
\widetilde{D}_{\xi\xi}^{\text{C}}=B_{t}(p)\end{array}\right.\]
 et \[
\left\{ \begin{array}{l}
\widetilde{D}_{p}^{\text{C}}=-F(p)\\
\widetilde{D}_{\xi}^{\text{C}}=0\end{array}\right.\]
 on obtient \[
\left\{ \begin{array}{l}
\widetilde{D}_{pp}^{\text{C}(0)}=\sigma\left\{ \frac{\sigma\xi}{\Psi(\theta)\xi_{0}}A(p)\right\} =\sigma\left\{ \frac{\sigma\xi}{\Psi(\theta)\xi_{0}}\right\} A(p)\\
\widetilde{D}_{p\xi}^{\text{C}(0)}=0\\
\widetilde{D}_{\xi p}^{\text{C}(0)}=0\\
\widetilde{D}_{\xi\xi}^{\text{C}(0)}=\sigma\left\{ \frac{\sigma\xi^{3}}{\Psi^{2}\xi_{0}^{3}}B_{t}(p)\right\} =\sigma\left\{ \frac{\sigma\xi^{3}}{\Psi^{2}\xi_{0}^{3}}\right\} B_{t}(p)\end{array}\right.\]
 et \[
\left\{ \begin{array}{l}
\widetilde{F}_{p}^{\text{C}(0)}=-\sigma\left\{ \frac{\sigma\xi}{\Psi(\theta)\xi_{0}}F(p)\right\} =-\sigma\left\{ \frac{\sigma\xi}{\Psi(\theta)\xi_{0}}\right\} F(p)\\
\widetilde{F}_{\xi}^{\text{C}(0)}=\sigma\frac{\sqrt{1-\xi_{0}^{2}}}{p\xi_{0}^{3}}\left\{ \frac{\sigma\xi\left(\Psi(\theta)-1\right)}{\Psi^{2}\xi_{0}}B_{t}(p)\right\} =\sigma\frac{\sqrt{1-\xi_{0}^{2}}}{p\xi_{0}^{3}}B_{t}(p)\left\{ \frac{\sigma\xi\left(\Psi(\theta)-1\right)}{\Psi^{2}\xi_{0}}\right\} \end{array}\right.\]


Il faut determiner les coefficients suivants: \begin{eqnarray*}
\left\{ \frac{\sigma\xi}{\Psi(\theta)\xi_{0}}\right\}  & = & \frac{\sigma}{\lambda}H(|\xi_{0}|-\xi_{0T})\int_{0}^{\pi}\frac{d\theta}{\pi}\frac{1}{\Psi}\\
 & = & \frac{\sigma}{\lambda\left(1+\epsilon\right)}H(|\xi_{0}|-\xi_{0T})\int_{0}^{\pi}\frac{d\theta}{\pi}\left(1+\epsilon\cos\theta\right)\\
 & = & \frac{\sigma}{\lambda\left(1+\epsilon\right)}H(|\xi_{0}|-\xi_{0T})\end{eqnarray*}
\begin{eqnarray*}
\left\{ \frac{\sigma\xi^{3}}{\Psi^{2}\xi_{0}^{3}}\right\}  & = & \frac{\sigma}{\lambda}H(|\xi_{0}|-\xi_{0T})\int_{0}^{\pi}\frac{d\theta}{\pi}\frac{\xi^{2}}{\Psi^{2}\xi_{0}^{2}}\\
 & = & \frac{\sigma}{\lambda}H(|\xi_{0}|-\xi_{0T})\int_{0}^{\pi}\frac{d\theta}{\pi}\frac{1-\Psi(1-\xi_{0}^{2})}{\Psi^{2}\xi_{0}^{2}}\\
 & = & \frac{\sigma}{\lambda}H(|\xi_{0}|-\xi_{0T})\int_{0}^{\pi}\frac{d\theta}{\pi}\left(\frac{1}{\Psi}-\frac{1}{\xi_{0}^{2}}\frac{\Psi-1}{\Psi^{2}}\right)\\
 & = & \frac{\sigma}{\lambda}H(|\xi_{0}|-\xi_{0T})\left[\frac{1}{\left(1+\epsilon\right)}-\frac{1}{\xi_{0}^{2}}\frac{\epsilon}{\left(1+\epsilon\right)^{2}}\int_{0}^{\pi}\frac{d\theta}{\pi}\left(1-\cos\theta\right)\left(1+\epsilon\cos\theta\right)\right]\\
 & = & \frac{\sigma}{\lambda\left(1+\epsilon\right)}H(|\xi_{0}|-\xi_{0T})\left[1-\frac{1}{\xi_{0}^{2}}\frac{\epsilon}{\left(1+\epsilon\right)}\left(1-\frac{\epsilon}{2}\right)\right]\end{eqnarray*}
\begin{eqnarray*}
\left\{ \frac{\sigma\xi\left(\Psi(\theta)-1\right)}{\Psi^{2}\xi_{0}}\right\}  & = & \frac{\sigma}{\lambda}H(|\xi_{0}|-\xi_{0T})\int_{0}^{\pi}\frac{d\theta}{\pi}\frac{\Psi(\theta)-1}{\Psi^{2}}\\
 & = & \frac{\sigma}{\lambda}\frac{\epsilon}{\left(1+\epsilon\right)^{2}}\left(1-\frac{\epsilon}{2}\right)\end{eqnarray*}


Finalement, on obtient \begin{eqnarray*}
\widetilde{D}_{pp}^{\text{C}(0)} & = & \frac{1}{\lambda\left(1+\epsilon\right)}H(|\xi_{0}|-\xi_{0T})A(p)\\
\widetilde{D}_{p\xi}^{\text{C}(0)} & = & 0\\
\widetilde{D}_{\xi p}^{\text{C}(0)} & = & 0\\
\widetilde{D}_{\xi\xi}^{\text{C}(0)} & = & \frac{1}{\lambda\left(1+\epsilon\right)}H(|\xi_{0}|-\xi_{0T})\left[1-\frac{1}{\xi_{0}^{2}}\frac{\epsilon}{\left(1+\epsilon\right)}\left(1-\frac{\epsilon}{2}\right)\right]B_{t}(p)\end{eqnarray*}
 et

\begin{eqnarray*}
\widetilde{F}_{p}^{\text{C}(0)} & = & -\frac{1}{\lambda\left(1+\epsilon\right)}H(|\xi_{0}|-\xi_{0T})F(p)\\
\widetilde{F}_{\xi}^{\text{C}(0)} & = & \frac{1}{\lambda}H(|\xi_{0}|-\xi_{0T})\frac{\epsilon\left(1-\epsilon/2\right)}{\left(1+\epsilon\right)^{2}}\frac{\sqrt{1-\xi_{0}^{2}}}{p\xi_{0}^{3}}B_{t}(p)\end{eqnarray*}


Sous forme discrete, les coefficients qui interviennent dans la matrice
$\widetilde{\overline{\overline{M}}}^{\left(0\right)}$ sont

\[
\widetilde{\overline{\overline{D}}}^{\text{C}(0)}\rightarrow\left\{ \begin{array}{l}
\widetilde{D}_{pp,i+1,j+1/2}^{\text{C}(0)}=\frac{1}{\lambda\left(\xi_{0j+1/2}\right)}\frac{1}{\left(1+\epsilon\right)}H(|\xi_{0,j+1/2}|-\xi_{0T})A_{i+1}\\
\widetilde{D}_{pp,i,j+1/2}^{\text{C}(0)}=\frac{1}{\lambda\left(\xi_{0j+1/2}\right)}\frac{1}{\left(1+\epsilon\right)}H(|\xi_{0,j+1/2}|-\xi_{0T})A_{i}\\
\widetilde{D}_{p\xi,i+3/2,j+1/2}^{\text{C}\left(0\right)}=0\\
\widetilde{D}_{\xi p,i+1/2,j+3/2}^{\text{C}\left(0\right)}=0\\
\widetilde{D}_{p\xi,i+1/2,j+1/2}^{\text{C}\left(0\right)}=0\\
\widetilde{D}_{\xi p,i+1/2,j+1/2}^{\text{C}\left(0\right)}=0\\
\widetilde{D}_{p\xi,i-1/2,j+1/2}^{\text{C}\left(0\right)}=0\\
\widetilde{D}_{\xi p,i+1/2,j-1/2}^{\text{C}\left(0\right)}=0\\
\widetilde{D}_{\xi\xi,i+1/2,j+1}^{\text{C}(0)}=\frac{1}{\lambda\left(\xi_{0j+1}\right)}\frac{1}{\left(1+\epsilon\right)}H(|\xi_{0,j+1}|-\xi_{0T})\left[1-\frac{1}{\xi_{0,j+1}^{2}}\frac{\epsilon}{\left(1+\epsilon\right)}\left(1-\frac{\epsilon}{2}\right)\right]B_{t,i+1/2}\\
\widetilde{D}_{\xi\xi,i+1/2,j}^{\text{C}(0)}=\frac{1}{\lambda\left(\xi_{0j}\right)}\frac{1}{\left(1+\epsilon\right)}H(|\xi_{0,j}|-\xi_{0T})\left[1-\frac{1}{\xi_{0,j}^{2}}\frac{\epsilon}{\left(1+\epsilon\right)}\left(1-\frac{\epsilon}{2}\right)\right]B_{t,i+1/2}\end{array}\right.\]
 et \[
\widetilde{\overline{\overline{F}}}^{\text{C}(0)}\rightarrow\left\{ \begin{array}{l}
\widetilde{F}_{p,i+1,j+1/2}^{\text{C}(0)}=-\frac{1}{\lambda\left(\xi_{0j+1/2}\right)}\frac{1}{\left(1+\epsilon\right)}H(|\xi_{0,j+1/2}|-\xi_{0T})F_{i+1}\\
\widetilde{F}_{p,i,j+1/2}^{\text{C}(0)}=-\frac{1}{\lambda\left(\xi_{0j+1/2}\right)}\frac{1}{\left(1+\epsilon\right)}H(|\xi_{0,j+1/2}|-\xi_{0T})F_{i}\\
\widetilde{F}_{\xi,i+1/2,j+1}^{\text{C}(0)}=\frac{1}{\lambda\left(\xi_{0j+1}\right)}\frac{\epsilon}{\left(1+\epsilon\right)^{2}}\left(1-\frac{\epsilon}{2}\right)\frac{\sqrt{1-\xi_{0,j+1}^{2}}}{p_{i+1/2}\xi_{0,j+1}^{3}}B_{t}(p)\\
\widetilde{F}_{\xi,i+1/2,j}^{\text{C}(0)}=\frac{1}{\lambda\left(\xi_{0j}\right)}\frac{\epsilon}{\left(1+\epsilon\right)^{2}}\left(1-\frac{\epsilon}{2}\right)\frac{\sqrt{1-\xi_{0,j}^{2}}}{p_{i+1/2}\xi_{0,j}^{3}}B_{t}(p)\end{array}\right.\]
