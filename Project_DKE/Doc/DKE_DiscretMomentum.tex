
The starting point of the discrete representation of the first order drift
kinetic equation is the conservative relation
\begin{eqnarray}
&&\left. \frac{\partial \left( p^{2}\mathbf{S}_{p}^{\left( 0\right) }\right)
}{\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k)}-\frac{p_{i+1/2}}{\lambda
^{l+1/2,j+1/2}}\times   \nonumber \\
&&\left. \frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}%
\lambda \mathbf{S}_{\xi }^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}^{(k)}  \nonumber \\
&=&\left. \frac{\partial \left( p^{2}\widetilde{\mathbf{S}}_{p}^{\left(
0\right) }\right) }{\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k)}-\frac{%
p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}\times   \nonumber \\
&&\left. \frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}%
\lambda \widetilde{\mathbf{S}}_{\xi }^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}^{(k)}
\end{eqnarray}
where $\mathbf{S}_{p}^{\left( 0\right) }$ and $\mathbf{S}_{\xi }^{\left(
0\right) }$ are fluxes related to the function $g^{\left( 0\right) }$ as
introduced in Sec.\ref{dkefluxconsmomentum}, while $\widetilde{\mathbf{S}}%
_{p}^{\left( 0\right) }$ and $\widetilde{\mathbf{S}}_{\xi }^{\left( 0\right)
}$ are fluxes related to the function $\widetilde{f}^{\left( 0\right) }$.
Since $g^{\left( 0\right) }$ and $f_{0}^{\left( 0\right) }$ have same
symmetries with respect to the pitch-angle $\xi _{0}$, matrix coefficients
are exactly identical for both functions (see Sec. \ref{fpdiscretemomentum}%
). However, calculations are slightly different for $\widetilde{f}^{\left(
0\right) },$ though a conservative form may still be kept. By analogy with
zero order Fokker-Planck equation,
\begin{eqnarray}
\frac{\partial \left( p^{2}\widetilde{S}_{p}^{\left( 0\right) }\right) }{%
\partial p} &=&\frac{\partial }{\partial p}\left( -p^{2}\widetilde{D}%
_{pp}^{\left( 0\right) }\frac{\partial \widetilde{f}^{\left( 0\right) }}{%
\partial p}+p^{2}\widetilde{F}_{p}^{\left( 0\right) }\widetilde{f}^{\left(
0\right) }\right)   \nonumber \\
&&+\sqrt{1-\xi _{0}^{2}}\frac{\partial }{\partial p}\left( p\widetilde{D}%
_{p\xi }^{\left( 0\right) }\right) \frac{\partial \widetilde{f}^{\left(
0\right) }}{\partial \xi _{0}}  \nonumber \\
&&+\sqrt{1-\xi _{0}^{2}}p\widetilde{D}_{p\xi }^{\left( 0\right) }\frac{%
\partial ^{2}\widetilde{f}^{\left( 0\right) }}{\partial p\partial \xi _{0}}
\end{eqnarray}
\begin{eqnarray}
\frac{\partial \left( \sqrt{1-\xi _{0}^{2}}\lambda \left( \psi ,\xi
_{0}\right) \widetilde{S}_{\xi }^{\left( 0\right) }\right) }{\partial \xi
_{0}} &=&\frac{\partial }{\partial \xi _{0}}\left( \widetilde{D}_{\xi \xi
}^{\left( 0\right) }\frac{1-\xi _{0}^{2}}{p}\lambda \left( \psi ,\xi
_{0}\right) \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial \xi
_{0}}\right.   \nonumber \\
&&\left. +\sqrt{1-\xi _{0}^{2}}\lambda \left( \psi ,\xi _{0}\right)
\widetilde{F}_{\xi }^{\left( 0\right) }\widetilde{f}^{\left( 0\right)
}\right)   \nonumber \\
&&-\frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}\lambda
\left( \psi ,\xi _{0}\right) \widetilde{D}_{\xi p}^{\left( 0\right) }\right)
\frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial p}  \nonumber \\
&&-\sqrt{1-\xi _{0}^{2}}\lambda \left( \psi ,\xi _{0}\right) \widetilde{D}%
_{\xi p}^{\left( 0\right) }\frac{\partial ^{2}\widetilde{f}^{\left( 0\right)
}}{\partial \xi _{0}\partial p}
\end{eqnarray}
one obtains
\begin{eqnarray}
&&\left. \frac{\partial \left( p^{2}\widetilde{\mathbf{S}}_{p}^{\left(
0\right) }\right) }{\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k)}-\frac{%
p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}\times   \nonumber \\
&&\left. \frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}%
\lambda \widetilde{\mathbf{S}}_{\xi }^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}^{(k)}  \nonumber \\
&=&\sum_{m=1}^{m=8}\widetilde{T}^{\left[ m\right] }
\end{eqnarray}
with
\begin{equation}
\widetilde{T}^{\left[ 1\right] }=\frac{-p_{i+1}^{2}\widetilde{D}%
_{pp,l+1/2,i+1,j+1/2}^{\left( 0\right) }\left. \frac{\partial \widetilde{f}%
^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1,j+1/2}^{(k+1)}+p_{i+1}^{2}\widetilde{F}_{p,l+1/2,i+1,j+1/2}^{%
\left( 0\right) }\widetilde{f}_{l+1/2,i+1,j+1/2}^{\left( 0\right) (k+1)}}{%
\Delta p_{i+1/2}}
\end{equation}
\begin{equation}
\widetilde{T}^{\left[ 2\right] }=\frac{p_{i}^{2}\widetilde{D}%
_{pp,l+1/2,i,j+1/2}^{\left( 0\right) }\left. \frac{\partial \widetilde{f}%
^{\left( 0\right) }}{\partial p}\right| _{l+1/2,i,j+1/2}^{(k+1)}-p_{i}^{2}%
\widetilde{F}_{p,l+1/2,i,j+1/2}^{\left( 0\right) }\widetilde{f}%
_{l+1/2,i,j+1/2}^{\left( 0\right) (k+1)}}{\Delta p_{i+1/2}}
\end{equation}
\begin{equation}
\widetilde{T}^{\left[ 3\right] }=+\sqrt{1-\xi _{0,j+1/2}^{2}}\left. \frac{%
\partial \left( p\widetilde{D}_{p\xi }^{\left( 0\right) }\right) }{\partial p%
}\right| _{l+1/2,i+1/2,j+1/2}\left. \frac{\partial \widetilde{f}^{\left(
0\right) }}{\partial \xi }\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}
\begin{equation}
\widetilde{T}^{\left[ 4\right] }=+\sqrt{1-\xi _{0,j+1/2}^{2}}p_{i+1/2}%
\widetilde{D}_{p\xi ,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }\left. \frac{%
\partial ^{2}\widetilde{f}^{\left( 0\right) }}{\partial p\partial \xi }%
\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}
\begin{eqnarray}
\widetilde{T}^{\left[ 5\right] } &=&-\frac{p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}%
\left[ \frac{\widetilde{D}_{\xi \xi ,l+1/2,i+1/2,j+1}^{\left( 0\right)
}\left( 1-\xi _{0,j+1}^{2}\right) \lambda ^{l+1/2,j+1}\left. \frac{\partial
\widetilde{f}^{\left( 0\right) }}{\partial \xi }\right|
_{l+1/2,i+1/2,j+1}^{(k+1)}}{p_{i+1/2}\Delta \xi _{j+1/2}}\right.   \nonumber
\\
&&+\left. \frac{p_{i+1/2}\sqrt{1-\xi _{0,j+1}^{2}}\lambda ^{l+1/2,j+1}%
\widetilde{F}_{\xi ,l+1/2,i+1/2,j+1}^{\left( 0\right) }\widetilde{f}%
_{l+1/2,i+1/2,j+1}^{\left( 0\right) (k+1)}}{\Delta \xi _{j+1/2}}\right]
\end{eqnarray}
\begin{eqnarray}
\widetilde{T}^{\left[ 6\right] } &=&\frac{p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}%
\left[ \frac{\widetilde{D}_{\xi \xi ,l+1/2,i+1/2,j}^{\left( 0\right) }\left(
1-\xi _{0,j}^{2}\right) \lambda ^{l+1/2,j}\left. \frac{\partial \widetilde{f}%
^{\left( 0\right) }}{\partial \xi }\right| _{l+1/2,i+1/2,j}^{(k+1)}}{%
p_{i+1/2}\Delta \xi _{j+1/2}}\right.   \nonumber \\
&&+\left. \frac{p_{i+1/2}\sqrt{1-\xi _{0,j}^{2}}\lambda ^{l+1/2,j}\widetilde{%
F}_{\xi ,l+1/2,i+1/2,j}^{\left( 0\right) }\widetilde{f}_{l+1/2,i+1/2,j}^{%
\left( 0\right) (k+1)}}{\Delta \xi _{j+1/2}}\right]
\end{eqnarray}

\begin{equation}
\widetilde{T}^{\left[ 7\right] }=\frac{p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}%
\left. \frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}%
\lambda \widetilde{D}_{\xi p}^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{%
\partial p}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}
\begin{equation}
\widetilde{T}^{\left[ 8\right] }=\frac{p_{i+1/2}}{\lambda ^{l+1/2,j+1/2}}%
\sqrt{1-\xi _{0,j+1/2}^{2}}\lambda ^{l+1/2,j+1/2}\widetilde{D}_{\xi
p,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }\left. \frac{\partial ^{2}\widetilde{f%
}^{\left( 0\right) }}{\partial \xi _{0}\partial p}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{equation}

Discrete expressions of the partial derivatives are,
\begin{equation}
\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1,j+1/2}^{(k+1)}=\frac{\widetilde{f}_{l+1/2,i+3/2,j+1/2}^{\left(
0\right) (k+1)}-\widetilde{f}_{l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{%
\Delta p_{i+1}}
\end{equation}
\begin{equation}
\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{\widetilde{f}_{l+1/2,i+3/2,j+1/2}^{\left(
0\right) (k+1)}-\widetilde{f}_{l+1/2,i-1/2,j+1/2}^{\left( 0\right) (k+1)}}{%
\Delta p_{i+1}+\Delta p_{i}}
\end{equation}
\begin{equation}
\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial p}\right|
_{l+1/2,i,j+1/2}^{(k+1)}=\frac{\widetilde{f}_{l+1/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}-\widetilde{f}_{l+1/2,i-1/2,j+1/2}^{\left( 0\right) (k+1)}}{%
\Delta p_{i}}
\end{equation}
\begin{equation}
\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial \xi _{0}}
\right| _{l+1/2,i+1/2,j+1}^{(k+1)}=\frac{\widetilde{f}_{l+1/2,i+1/2,j+3/2}^{%
\left( 0\right) (k+1)}-\widetilde{f}_{l+1/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}}{\Delta \xi _{0,j+1}}
\end{equation}
\begin{equation}
\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial \xi _{0}}%
\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{\widetilde{f}%
_{l+1/2,i+1/2,j+3/2}^{\left( 0\right) (k+1)}-\widetilde{f}
_{l+1/2,i+1/2,j-1/2}^{\left( 0\right) (k+1)}}{\Delta \xi _{0,j+1}+\Delta \xi
_{0,j}}
\end{equation}
\begin{equation}
\left. \frac{\partial \widetilde{f}^{\left( 0\right) }}{\partial \xi _{0}}
\right| _{l+1/2,i+1/2,j}^{(k+1)}=\frac{\widetilde{f}_{l+1/2,i+1/2,j+1/2}^{%
\left( 0\right) (k+1)}-\widetilde{f}_{l+1/2,i+1/2,j-1/2}^{\left( 0\right)
(k+1)}}{\Delta \xi _{0,j}}
\end{equation}
and cross-derivatives
\begin{eqnarray}
\left. \frac{\partial ^{2}\widetilde{f}^{\left( 0\right) }}{\partial
p\partial \xi _{0}}\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)} &=&\left. \frac{%
\partial ^{2}\widetilde{f}^{\left( 0\right) }}{\partial \xi _{0}\partial p}
\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}  \nonumber \\
&=&\frac{\widetilde{f}_{l+1/2,i+3/2,j+3/2}^{\left( 0\right) (k+1)}+%
\widetilde{f}_{l+1/2,i-1/2,j-1/2}^{\left( 0\right) (k+1)}}{\left( \Delta
p_{i+1}+\Delta p_{i}\right) \left( \Delta \xi _{0,j+1}+\Delta \xi
_{0,j}\right) }  \nonumber \\
&&-\frac{\widetilde{f}_{l+1/2,i+3/2,j-1/2}^{\left( 0\right) (k+1)}+%
\widetilde{f}_{l+1/2,i-1/2,j+3/2}^{\left( 0\right) (k+1)}}{\left( \Delta
p_{i+1}+\Delta p_{i}\right) \left( \Delta \xi _{0,j+1}+\Delta \xi
_{0,j}\right) }
\end{eqnarray}

As for the zero-order Fokker-Planck equation, other derivatives in discrete
form become
\begin{equation}
\left. \frac{\partial \left( p\widetilde{D}_{p\xi }^{\left( 0\right)
}\right) }{\partial p}\right| _{l+1/2,i+1/2,j+1/2}=\frac{p_{i+1}\widetilde{D}%
_{p\xi ,l+1/2,i+1,j+1/2}^{\left( 0\right) }-p_{i}\widetilde{D}_{p\xi
,l+1/2,i,j+1/2}^{\left( 0\right) }}{\Delta p_{i+1/2}}
\end{equation}
\begin{eqnarray}
&&\left. \frac{\partial }{\partial \xi _{0}}\left( \sqrt{1-\xi _{0}^{2}}%
\lambda \widetilde{D}_{\xi p}^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}  \nonumber \\
&=&\frac{\sqrt{1-\xi _{0,j+1}^{2}}\lambda ^{l+1/2,j+1}\widetilde{D}_{\xi
p,l+1/2,i+1/2,j+1}^{\left( 0\right) }}{\Delta \xi _{0,j+1/2}}  \nonumber \\
&&-\frac{\sqrt{1-\xi _{0,j}^{2}}\lambda ^{l+1/2,j}\widetilde{D}_{\xi
p,l+1/2,i+1/2,j}^{\left( 0\right) }}{\Delta \xi _{0,j+1/2}}
\end{eqnarray}

Since the distribution function $\widetilde{f}^{\left( 0\right) }$is defined
on the half grid, while fluxes on the full grid, it is necessary to
interpolate, because in some derivatives, values of are taken on the full
grid. As discussed in the previous section, interpolation procedure is more
complex for $\widetilde{f}^{\left( 0\right) }$ than for $f_{0}^{\left(
0\right) }.$ Therefore, for terms proportional to $\widetilde{D}_{\xi \xi
}^{\left( 0\right) }$ and $\widetilde{F}_{\xi }^{\left( 0\right) }$%
\begin{eqnarray}
\widetilde{f}_{l+1/2,i+1,j+1/2}^{\left( 0\right) (k+1)} &=&\frac{p_{i+1}}{%
p_{i+3/2}}\left( 1-\delta _{p,l+1/2,i+1,j+1/2}^{(0)}\right) \widetilde{f}%
_{l+1/2,i+3/2,j+1/2}^{(0)(k+1)}  \nonumber \\
&&+\frac{p_{i+1}}{p_{i+1/2}}\delta _{p,l+1/2,i+1,j+1/2}^{(0)}\widetilde{f}%
_{l+1/2,i+1/2,j+1/2}^{(0)(k+1)}  \nonumber \\
&&+\widetilde{h}_{l+1/2,i+1,j+1/2}^{\left( 0\right) (k+1)}
\end{eqnarray}
\begin{eqnarray}
\widetilde{f}_{l+1/2,i,j+1/2}^{\left( 0\right) (k+1)} &=&\frac{p_{i}}{%
p_{i+1/2}}\left( 1-\delta _{p,l+1/2,i,j+1/2}^{(0)}\right) \widetilde{f}%
_{l+1/2,i+1/2,j+1/2}^{(0)(k+1)}  \nonumber \\
&&+\frac{p_{i}}{p_{i-1/2}}\delta _{p,l+1/2,i,j+1/2}^{(0)}\widetilde{f}
_{l+1/2,i-1/2,j+1/2}^{(0)(k+1)}  \nonumber \\
&&+\widetilde{h}_{l+1/2,i,j+1/2}^{\left( 0\right) (k+1)}
\end{eqnarray}
and for terms proportional to $\widetilde{D}_{\xi \xi }^{\left( 0\right) }$
and $\widetilde{F}_{\xi }^{\left( 0\right) }$

\begin{eqnarray}
\widetilde{f}_{l+1/2,i+1/2,j+1}^{\left( 0\right) (k+1)} &=&\frac{\xi _{0,j+1}%
}{\xi _{0,j+3/2}}\left( 1-\delta _{\xi ,l+1/2,i+1/2,j+1}^{(0)}\right)
\widetilde{f}_{l+1/2,i+1/2,j+3/2}^{(0)(k+1)}  \nonumber \\
&&+\frac{\xi _{0,j+1}}{\xi _{0,j+1/2}}\delta _{\xi ,l+1/2,i+1/2,j+1}^{(0)}%
\widetilde{f}_{l+1/2,i+1/2,j+1/2}^{(0)(k+1)}
\end{eqnarray}
\begin{eqnarray}
\widetilde{f}_{l+1/2,i+1/2,j}^{\left( 0\right) (k+1)} &=&\frac{\xi _{0,j}}{%
\xi _{0,j+1/2}}\left( 1-\delta _{\xi ,l+1/2,i+1/2,j}^{(0)}\right) \widetilde{%
f}_{l+1/2,i+1/2,j+1/2}^{(0)(k+1)}  \nonumber \\
&&+\frac{\xi _{0,j}}{\xi _{0,j-1/2}}\delta _{\xi ,l+1/2,i+1/2,j}^{(0)}%
\widetilde{f}_{l+1/2,i+1/2,j-1/2}^{(0)(k+1)}
\end{eqnarray}
since $\widetilde{h}_{l+1/2,i+1/2,j+1}^{\left( 0\right) }=\widetilde{h}%
_{l+1/2,i+1/2,j}^{\left( 0\right) }=0.$

Gathering all terms in a matrix form

where $\widetilde{\overline{M}}^{\left( 0\right) }$ and $\overline{\overline{%

\widetilde{\overline{H}}_{\psi }^{\left( 0\right) }$ that results from the

\widetilde{\overline{M}}_{p,l+1/2,i+3/2,j+3/2}^{\left( 0\right) } &=&\frac{%

,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\

_{0,j+1/2}^{2}}}{\Delta \xi _{0,j+1}+\Delta \xi _{0,j}}\widetilde{D}_{\xi
p,l+1/2,i+1/2,j+1/2}^{\left( 0\right) }\nonumber \\
\end{eqnarray}
\begin{eqnarray}
\widetilde{\overline{M}}_{p,l+1/2,i+1/2,j+3/2}^{\left( 0\right) } &=&\frac{%
p_{i+1}}{\Delta p_{i+1/2}}\frac{\sqrt{1-\xi _{0,j+1/2}^{2}}}{\Delta \xi
_{0,j+1}+\Delta \xi _{0,j}}\widetilde{D}_{p\xi ,l+1/2,i+1,j+1/2}^{\left(
0\right) }  \nonumber \\
&&-\frac{p_{i}}{\Delta p_{i+1/2}}\frac{\sqrt{1-\xi _{0,j+1/2}^{2}}}{\Delta %
\xi _{0,j+1}+\Delta \xi _{0,j}}\widetilde{D}_{p\xi ,l+1/2,i,j+1/2}^{\left(
0\right) }  \nonumber \\
&&-\frac{1-\xi _{0,j+1}^{2}}{\Delta \xi _{0,j+1/2}\Delta \xi _{0,j+1}}\frac{%

\end{eqnarray}
\begin{eqnarray}

\begin{eqnarray}

_{pp,l+1/2,i+1,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi
_{0,j+1}^{2}}}{\Delta \xi _{0,j+1/2}}\times   \nonumber \\
&&\frac{\lambda ^{l+1/2,j+1}}{\lambda ^{l+1/2,j+1/2}}\widetilde{D}_{\xi
p,l+1/2,i+1/2,j+1}^{\left( 0\right) }  \nonumber \\
&&-\frac{p_{i+1/2}}{\Delta p_{i+1}+\Delta p_{i}}\frac{\sqrt{1-\xi _{0,j}^{2}}%
}{\Delta \xi _{0,j+1/2}}\times   \nonumber \\
