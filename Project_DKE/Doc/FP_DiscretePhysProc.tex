\subsection{Collisions}

According to the bounce averaged expression,
\begin{equation}
\overline{\overline{D}}_{p}^{C\left( 0\right) }\rightarrow \left\{
\begin{array}{l}
D_{pp,l+1/2,i+1,j+1/2}^{C\left( 0\right) }=A_{l+1/2,i+1} \\
D_{pp,l+1/2,i,j+1/2}^{C\left( 0\right) }=A_{l+1/2,i} \\
D_{p\xi ,l+1/2,i+1,j+1/2}^{C\left( 0\right) }=0 \\
D_{\xi p,l+1/2,i+1/2,j+1}^{C\left( 0\right) }=0 \\
D_{p\xi ,l+1/2,i+1/2,j+1/2}^{C\left( 0\right) }=0 \\
D_{\xi p,l+1/2,i+1/2,j+1/2}^{C\left( 0\right) }=0 \\
D_{p\xi ,l+1/2,i,j+1/2}^{C\left( 0\right) }=0 \\
D_{\xi p,l+1/2,i+1/2,j}^{C\left( 0\right) }=0 \\
D_{\xi \xi ,l+1/2,i+1/2,j+1}^{C\left( 0\right) }=\left( \lambda
_{2,-1,0}^{l+1/2,j+1}/\lambda ^{l+1/2,j+1}\right) B_{t,l+1/2,i+1/2} \\
D_{\xi \xi ,l+1/2,i+1/2,j}^{C\left( 0\right) }=\left( \lambda
_{2,-1,0}^{l+1/2,j}/\lambda ^{l+1/2,j}\right) B_{t,l+1/2,i+1/2}
\end{array}
\right.
\end{equation}
and
\begin{equation}
\overline{\overline{F}}_{p}^{C\left( 0\right) }\rightarrow \left\{
\begin{array}{l}
F_{p,l+1/2,i+1,j+1/2}^{C\left( 0\right) }=-F_{l+1/2,i+1} \\
F_{p,l+1/2,i,j+1/2}^{C\left( 0\right) }=-F_{l+1/2,i} \\
F_{\xi ,l+1/2,i+1/2,j+1}^{C\left( 0\right) }=0 \\
F_{\xi ,l+1/2,i+1/2,j}^{C\left( 0\right) }=0
\end{array}
\right.
\end{equation}
where each coefficient is the sum of the electron-electron and electron-ion
interactions.

\subsubsection{Electron-electron collision operator}

\paragraph{Belaiev-Budker relativistic collision model}

Here, coefficients corresponding to the Belaiev-Budker collision operator
that ranges from non-relativistic to fully relativistic regimes may be
expressed as, according to the notation used in Ref.\cite{shk97},
\begin{equation}
A_{l+1/2,i+1/2}^{ee}=\frac{\left( F_{1,l+1/2,i+1/2}+F_{2,l+1/2,i+1/2}\right)
}{v_{i+1/2}}\overline{T}_{e,l+1/2}
\end{equation}
\begin{equation}
F_{l+1/2,i+1/2}^{ee}=F_{1,l+1/2,i+1/2}+F_{2,l+1/2,i+1/2}
\end{equation}
with
\begin{equation}
F_{1,l+1/2,i+1/2}=\frac{4\pi }{v_{i+1/2}^{2}}F_{11,l+1/2,i+1/2}+\frac{4\pi }{%
p_{i+1/2}^{2}}F_{12,l+1/2,i+1/2}
\end{equation}
\begin{equation}
F_{2,l+1/2,i+1/2}=\frac{4\pi }{v_{i+1/2}}\left( 1-\frac{\gamma _{i+1/2}\zeta
_{,i+1/2}}{z_{i+1/2}}\right) F_{21,l+1/2,i+1/2}
\end{equation}

Expressions of coefficients $F_{11,l+1/2,i+1/2},$ $F_{12,l+1/2,i+1/2}$ and $%
F_{21,l+1/2,i+1/2}$ are

\begin{equation}
F_{11,l+1/2,i+1/2}=\int_{0}^{i+1/2}p^{\prime }v^{\prime }f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
\begin{equation}
F_{12,l+1/2,i+1/2}=\int_{0}^{i+1/2}p^{\prime }v^{\prime }\left( 1-\frac{%
\gamma ^{\prime }\zeta ^{\prime }}{z^{\prime }}\right) f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
\begin{equation}
F_{21,l+1/2,i+1/2}=\int_{i+1/2}^{\infty }p^{\prime }f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}

Coefficients $A^{ee}$ and $F^{ee}$ must be also calculated on the flux grids
and therefore, according to the previous definition,

\begin{equation}
A_{l+1/2,i}^{ee}=\frac{\left( F_{1,l+1/2,i}+F_{2,l+1/2,i}\right) }{v_{i}}%
\overline{T}_{e,l+1/2}
\end{equation}
and
\begin{equation}
F_{l+1/2,i}^{ee}=F_{1,l+1/2,i}+F_{2,l+1/2,i}
\end{equation}
where
\begin{equation}
F_{1,l+1/2,i}=\frac{4\pi }{v_{i}^{2}}F_{11,l+1/2,i}+\frac{4\pi }{p_{i}^{2}}%
F_{12,l+1/2,i}
\end{equation}
\begin{equation}
F_{2,l+1/2,i}=\frac{4\pi }{v_{i}}\left( 1-\frac{\gamma _{i}\zeta _{i}}{z_{i}}%
\right) F_{21,l+1/2,i}
\end{equation}
and
\begin{equation}
F_{11,l+1/2,i}=\int_{0}^{i}p^{\prime }v^{\prime }f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
\begin{equation}
F_{12,l+1/2,i}=\int_{0}^{i}p^{\prime }v^{\prime }\left( 1-\frac{\gamma
^{\prime }\zeta ^{\prime }}{z^{\prime }}\right) f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
\begin{equation}
F_{21,l+1/2,i}=\int_{i}^{\infty }p^{\prime }f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
for the grid points $i,$ while for the grid points $i+1,$ one has just to
replace $i$ by $i+1$ in the set of above expressions.

Furthermore, the expression of coefficient $B_{t}^{ee}$ is
\begin{equation}
B_{t,l+1/2,i+1/2}^{ee}=B_{t1,l+1/2,i+1/2}+B_{t2,l+1/2,i+1/2}
\end{equation}
with
\begin{equation}
B_{t1,l+1/2,i+1/2}=4\pi \sum_{n=1}^{5}B_{t1,l+1/2,i+1/2}^{\left[ n\right] }
\end{equation}
and
\begin{equation}
B_{t2,l+1/2,i+1/2}=4\pi \sum_{n=1}^{5}B_{t2,l+1/2,i+1/2}^{\left[ n\right] }
\end{equation}
where
\begin{equation}
B_{t1,l+1/2,i+1/2}^{\left[ 1\right] }=\frac{1}{2v_{i+1/2}}%
\int_{0}^{i+1/2}p^{\prime 2}f_{M}\left( \psi _{l+1/2},p^{\prime }\right)
dp^{\prime }
\end{equation}
\begin{equation}
B_{t1,l+1/2,i+1/2}^{\left[ 2\right] }=-\frac{1}{6v_{i+1/2}p_{i+1/2}^{2}}%
\int_{0}^{i+1/2}p^{\prime 4}f_{M}\left( \psi _{l+1/2},p^{\prime }\right)
dp^{\prime }
\end{equation}
\begin{equation}
B_{t1,l+1/2,i+1/2}^{\left[ 3\right] }=\frac{1}{8\gamma
_{i+1/2}^{2}z_{i+1/2}^{2}}\int_{0}^{i+1/2}p^{\prime 2}\frac{J_{1}^{\prime }}{%
\gamma ^{\prime }}f_{M}\left( \psi _{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
\begin{equation}
B_{t1,l+1/2,i+1/2}^{\left[ 4\right] }=-\frac{1}{4z_{i+1/2}^{2}}%
\int_{0}^{i+1/2}p^{\prime 2}\frac{J_{2}^{\prime }}{\gamma ^{\prime }}%
f_{M}\left( \psi _{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}

\begin{equation}
B_{t1,l+1/2,i+1/2}^{\left[ 5\right] }=-\frac{1}{4\gamma _{i+1/2}^{2}}
\int_{0}^{i+1/2}\frac{p^{\prime 2}}{\gamma ^{\prime }}\left( \gamma ^{\prime
}-\frac{\zeta ^{\prime }}{z^{\prime }}\right) f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
and
\begin{equation}
B_{t2,l+1/2,i+1/2}^{\left[ 1\right] }=\frac{1}{2}\int_{i+1/2}^{\infty }\frac{
p^{\prime 2}}{v^{\prime }}f_{M}\left( \psi _{l+1/2},p^{\prime }\right)
dp^{\prime }
\end{equation}
\begin{equation}
B_{t2,l+1/2,i+1/2}^{\left[ 2\right] }=-\frac{\gamma _{i+1/2}^{2}}{6}
\int_{i+1/2}^{\infty }\frac{p^{\prime 2}}{\gamma ^{\prime 2}v^{\prime }}%
f_{M}\left( \psi _{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}
\begin{equation}
B_{t2,l+1/2,i+1/2}^{\left[ 3\right] }=\frac{J_{1,i+1/2}}{8\gamma
_{i+1/2}z_{i+1/2}^{2}}\int_{i+1/2}^{\infty }\frac{p^{\prime 2}}{v^{\prime }}%
\frac{1}{\gamma ^{\prime 2}}f_{M}\left( \psi _{l+1/2},p^{\prime }\right)
dp^{\prime }
\end{equation}
\begin{equation}
B_{t2,l+1/2,i+1/2}^{\left[ 4\right] }=-\frac{\gamma _{i+1/2}J_{2,i+1/2}}{
4z_{i+1/2}^{2}}\int_{i+1/2}^{\infty }\frac{p^{\prime 2}}{v^{\prime }}\frac{1%
}{\gamma ^{\prime 2}}f_{M}\left( \psi _{l+1/2},p^{\prime }\right) dp^{\prime
}
\end{equation}

\begin{equation}
B_{t2,l+1/2,i+1/2}^{\left[ 5\right] }=-\frac{1}{4\gamma _{i+1/2}p_{i+1/2}^{2}%
}\left( \gamma _{i+1/2}-\frac{\zeta _{i+1/2}}{z_{i+1/2}}\right)
\int_{i+1/2}^{\infty }p^{\prime 2}v^{\prime }f_{M}\left( \psi
_{l+1/2},p^{\prime }\right) dp^{\prime }
\end{equation}

Here,
\begin{equation}
J_{1,i+1/2}=-3\gamma _{i+1/2}+\zeta _{i+1/2}\left( \frac{3}{z_{i+1/2}}
+2z_{i+1/2}\right)
\end{equation}
\begin{equation}
J_{2,i+1/2}=\gamma _{i+1/2}+\frac{\zeta _{i+1/2}}{z_{i+1/2}}-\frac{2}{3}%
\gamma _{i+1/2}z_{i+1/2}^{2}
\end{equation}
with
\begin{equation}
z_{i+1/2}=\beta _{th}^{\dag 2}p_{i+1/2}
\end{equation}
\begin{equation}
\gamma _{i+1/2}=\sqrt{1+z_{i+1/2}^{2}}
\end{equation}
and
\begin{equation}
\zeta _{i+1/2}=\sinh ^{-1}z_{i+1/2}
\end{equation}

Coefficients $A,$ $B_{t}$ and $F$ for the Beliaev-Budker relativistic
collision operator need to calculate accurately integrals of type
\begin{equation}
\left\{
\begin{array}{l}
\int_{0}^{p_{i}}Xdp^{\prime } \\
\int_{0}^{p_{i+1/2}}Xdp^{\prime } \\
\int_{0}^{p_{i+1}}Xdp^{\prime }
\end{array}
\right.
\end{equation}
which require a special attention in the vicinity of $p=0$, and also of
type,
\begin{equation}
\left\{
\begin{array}{l}
\int_{p_{i}}^{\infty }Xdp^{\prime } \\
\int_{p_{i+1/2}}^{\infty }Xdp^{\prime } \\
\int_{p_{i+1}}^{\infty }Xdp^{\prime }
\end{array}
\right.
\end{equation}

The goal is to ensure an acceptable numerical accuracy which preserves the
conservative nature of the equations to be solved by numerical method,
without any use of ad-hoc boundary conditions to compensate spurious
particle leak arisinf from an improper flux balance at each grid point. For
this purpose, a new momentum grid called ``super-grid'' is introduced,
correponding to a refined mesh. Since no specific condition is required as
compared to the links between distribution and flux grids, the momentum
super-grid is simply defined as a sum of two different meshes for integrals
of type $\int_{0}^{p_{i+1/2}}Xdp^{\prime }$%
\begin{equation}
\left\{
\begin{array}{l}
p_{i^{\prime }}^{s}\in \left[ \frac{p_{1/2}}{n_{p}^{s}},p_{1/2}\right[
,i^{\prime }\rightarrow \left\{ 0,n_{p}^{s}-1\right\} \\
p_{i^{\prime }}^{s}\in \left[ p_{1/2},p_{i+1/2}\right] ,i^{\prime
}\rightarrow \left\{ 0,n_{p}^{s}-1\right\}
\end{array}
\right.
\end{equation}
one for very fine calculations below $p_{1/2},$ and a second, less refined
up to $p_{i+1/2}.$ For integrals of type $\int_{0}^{p_{i}}Xdp^{\prime }$ and
$\int_{0}^{p_{i+1}}Xdp^{\prime },$ the corresponding super-grids are defined
in the same way,
\begin{equation}
\left\{
\begin{array}{l}
p_{i^{\prime }}^{s}\in \left[ \frac{p_{1/2}}{n_{p}^{s}},p_{1}\right]
,i^{\prime }\rightarrow \left\{ 0,n_{p}^{s}-1\right\} \\
p_{i^{\prime }}^{s}\in \left[ p_{1},p_{i}\right] ,i^{\prime }\rightarrow
\left\{ 0,n_{p}^{s}-1\right\}
\end{array}
\right.
\end{equation}
and
\begin{equation}
\left\{
\begin{array}{l}
p_{i^{\prime }}^{s}\in \left[ \frac{p_{1/2}}{n_{p}^{s}},p_{1}\right]
,i^{\prime }\rightarrow \left\{ 0,n_{p}^{s}-1\right\} \\
p_{i^{\prime }}^{s}\in \left[ p_{1},p_{i+1}\right] ,i^{\prime }\rightarrow
\left\{ 0,n_{p}^{s}-1\right\}
\end{array}
\right.
\end{equation}

It is important to note that $p_{1}$ correponds to the second point of the
grid $p_{i}$ while it is the first one for the grid $p_{i+1},$ so that
numerical integration starts at the same momentum value. Much in the same
way, integrals of type $\int_{p_{i}}^{\infty }Xdp^{\prime }$ , $%
\int_{p_{i+1/2}}^{\infty }Xdp^{\prime }$ and $\int_{p_{i+1}}^{\infty
}Xdp^{\prime }$ are simply defined on the following super-grids
\begin{equation}
\left\{
\begin{array}{l}
p_{i^{\prime }}^{s}\in \left[ p_{i},p_{\max }-\Delta p_{n_{p}}\right]
,i^{\prime }\rightarrow \left\{ 0,n_{p}^{s}-1\right\} \\
p_{i^{\prime }}^{s}\in \left[ p_{i+1/2},p_{\max }-\frac{\Delta p_{n_{p}-1/2}%
}{2}\right] ,i^{\prime }\rightarrow \left\{ 0,n_{p}^{s}-1\right\} \\
p_{i^{\prime }}^{s}\in \left[ p_{i+1},p_{\max }\right] ,i^{\prime
}\rightarrow \left\{ 0,n_{p}^{s}-1\right\}
\end{array}
\right.
\end{equation}

All integrals are performed by the trapezoidal method, even if any more
accurate technique like the Simpson method may be used in that case. A
crucial point is that integrals exactly end or start at points $p_{i},$ $%
p_{i+1/2}$ or $p_{i+1}$ so that no overlap between $%
\int_{0}^{p_{i+1/2}}Xdp^{\prime }$ and $\int_{p_{i+1/2}}^{\infty
}Xdp^{\prime }$ can take place. This is especially important to avoid
spurious numerical particle leak, that could break the conservative nature
of the equations to be solved.

Concerning the first order Legendre correction that ensures momentum
conservation, one must calculate
\begin{equation}
\left\{ C\left( f_{M},f\right) \right\} \simeq -\frac{3}{2}\frac{\overline{%
\lambda }_{1,1,0}}{\lambda }\xi _{0}\mathcal{I}\left( f_{M},f_{0}^{\left(
0\right) \left( m=1\right) }\right)
\end{equation}
on the distribution function grid, where
\begin{equation}
f_{0}^{\left( 0\right) \left( m=1\right) }\left( p,\xi _{0},\psi \right)
=\int_{-1}^{+1}\xi _{0}f_{0}^{\left( 0\right) }\left( p,\xi _{0},\psi
\right) d\xi _{0}
\end{equation}
as shown in Sec. \ref{FPCollisions}.

Therefore, by definition,
\begin{equation}
\left\{ C\left( f_{M},f\right) \right\} _{l+1/2,i+1/2,j+1/2}^{\left(
k\right) }=-\frac{3}{2}\frac{\overline{\lambda }_{1,1,0}^{l+1/2,j+1/2}}{
\lambda ^{l+1/2,j+1/2}}\xi _{0,j+1/2}\mathcal{I}\left( f_{M,},f_{0}^{\left(
0\right) \left( m=1\right) }\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }
\end{equation}
where for the Belaiev-Budker relativistic collision operator,

\begin{eqnarray}
\mathcal{I}&&\left( f_{0M,},f_{0}^{\left( 0\right) \left(
m=1\right) }\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }
\nonumber
\\&&=\frac{4\pi }{\gamma _{i+1/2}} \left(
f_{0,l=1,l+1/2,i+1/2,j+1/2}^{\left( 0\right) \left( k\right)
}\right.
\nonumber \\
&&+\frac{1}{p_{i+1/2}}\mathcal{I}_{1}\left( f_{M,},f_{0}^{\left( 0\right)
\left( m=1\right) }\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }  \nonumber
\\
&&\left. +p_{i+1/2}\mathcal{I}_{2}\left( f_{M,},f_{0}^{\left( 0\right)
\left( m=1\right) }\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }\right)
\end{eqnarray}
with
\begin{equation}
\mathcal{I}_{1}\left( f_{M,},f_{0}^{\left( 0\right) \left( m=1\right)
}\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }=\sum_{n=1}^{10}\mathcal{I}%
_{1,l+1/2,i+1/2,j+1/2}^{\left[ n\right] \left( k\right) }
\end{equation}
and
\begin{equation}
\mathcal{I}_{2}\left( f_{M,},f_{0}^{\left( 0\right) \left( m=1\right)
}\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }=\sum_{n=1}^{7}\mathcal{I}%
_{2,l+1/2,i+1/2,j+1/2}^{\left[ n\right] \left( k\right) }
\end{equation}

The set of coefficients $\mathcal{I}_{1}^{\left[ n\right] }$ is

\begin{equation}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 1\right] \left( k\right) }=\frac{1%
}{3\overline{T}_{e,l+1/2}}\sum_{i^{\prime }=0}^{_{i}}\frac{p_{i^{\prime
}+1/2}^{3}}{\gamma _{i^{\prime }+1/2}}f_{0,l+1/2,i^{\prime }+1/2}^{\left(
0\right) \left( m=1\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 2\right] \left( k\right) }=-\frac{
2\gamma _{i+1/2}}{3\overline{T}_{e,l+1/2}}\sum_{i^{\prime
}=0}^{_{i}}p_{i^{\prime }+1/2}^{3}f_{0,,l+1/2,i^{\prime }+1/2}^{\left(
0\right) \left( m=1\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 3\right] \left( k\right) }=\frac{
\gamma _{i+1/2}}{5\overline{T}_{e,l+1/2}^{2}}\sum_{i^{\prime }=0}^{_{i}}%
\frac{p_{i^{\prime }+1/2}^{5}}{\gamma _{i^{\prime }+1/2}}f_{0,l+1/2,i^{%
\prime }+1/2}^{\left( 0\right) \left( m=1\right) \left( k\right) }\Delta
p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 4\right] \left( k\right)
}=\sum_{i^{\prime }=0}^{_{i}}\frac{p_{i^{\prime }+1/2}}{\gamma _{i^{\prime
}+1/2}}\left( \gamma _{i^{\prime }+1/2}-\frac{\zeta _{i^{\prime }+1/2}}{
z_{i^{\prime }+1/2}}\right) f_{0,l+1/2,i^{\prime }+1/2}^{\left( 0\right)
\left( m=1\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 5\right] \left( k\right) }=-\frac{
\gamma _{i+1/2}}{\overline{T}_{e,l+1/2}}\sum_{i^{\prime }=0}^{i}\frac{%
p_{i^{\prime }+1/2}^{3}}{\gamma _{i^{\prime }+1/2}}\frac{J_{2,i^{\prime
}+1/2}}{z_{i^{\prime }+1/2}^{2}}f_{0,l+1/2,i^{\prime }+1/2}^{\left( 0\right)
\left( m=1\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}
\begin{eqnarray}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 6\right] \left( k\right) } &=&%
\frac{\gamma _{i+1/2}p_{i+1/2}^{2}-5\overline{T}_{e,l+1/2}}{6\overline{T}
_{e,l+1/2}^{2}}\sum_{i^{\prime }=0}^{_{i}}\frac{p_{i^{\prime }+1/2}^{3}}{%
\gamma _{i^{\prime }+1/2}}f_{0,l+1/2,i^{\prime }+1/2}^{\left( 0\right)

&&\times \left( 1+\frac{3}{z_{i^{\prime }+1/2}^{2}}-\frac{3\gamma

\begin{equation}

}+1/2}}\frac{J_{3,i^{\prime }+1/2}}{z_{i^{\prime }+1/2}}f_{0,l+1/2,i^{\prime

\begin{equation}

\left( m=1\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
\mathcal{I}_{1,l+1/2,i+1/2,j+1/2}^{\left[ 9\right] \left( k\right) }=\frac{

\paragraph{Relativistic Maxwellian background}

The relativistic Maxwellian background corresponds to that case where the
first order Legendre correction for momentum conservation is neglected.
Matrix coeffients $A^{ee},$ $B_{t}^{ee}$ and $F^{ee}$ determined in the
previous section are used.

\paragraph{Non-relativistic Maxwellian background}

For this case, matrix coefficients are

\begin{equation}
A_{l+1/2,i+1/2}^{ee}=\frac{\overline{n}_{e,l+1/2}}{%
2v_{i+1/2}u_{l+1/2,i+1/2}^{2}}\left[ erf \left( u_{l+1/2,i+1/2}\right)
-u_{l+1/2,i+1/2}erf ^{\prime }\left( u_{l+1/2,i+1/2}\right) \right]
\end{equation}

\begin{equation}
F_{l+1/2,i+1/2}^{ee}=\frac{\overline{n}_{e,l+1/2}}{v_{i+1/2}^{2}}\left[ erf
\left( u_{l+1/2,i+1/2}\right) -u_{l+1/2,i+1/2}erf ^{\prime }\left(
u_{l+1/2,i+1/2}\right) \right]
\end{equation}
and

\begin{align}
&B_{t,l+1/2,i+1/2}^{ee}=\frac{\overline{n}_{e,l+1/2}}{%
4v_{i+1/2}u_{l+1/2,i+1/2}^{2}}\left[ \left(
2u_{l+1/2,i+1/2}^{2}-1\right) erf \left( u_{l+1/2,i+1/2}\right)
\right.\nonumber\\&\qquad\left.+u_{l+1/2,i+1/2}erf ^{\prime
}\left( u_{l+1/2,i+1/2}\right) \right]
\end{align}
where
\begin{equation}
u_{l+1/2,i+1/2}=\frac{v_{i+1/2}}{2\overline{n}_{e,l+1/2}}
\end{equation}

For values of the matrix coefficients $A^{ee},$ $B_{t}^{ee}$ and $F^{ee}$ on
the momentum flux grids, one just has to replace $i+1/2$ by $i$ or $i+1$.

\paragraph{High velocity limit}

Though the high velocity limit corresponds to a restricted range of
application regarding the full collision operator, it can contribute to
useful comparisons with some theoretical calculations. Therefore, this
possibility has been implemented in the code. In that case, expressions of
the coefficients are greatly simplified,

\begin{equation}
A_{l+1/2,i+1/2}^{ee}=\frac{1}{v_{i+1/2}^{3}}\overline{n}_{e,l+1/2}\overline{T%
}_{e,l+1/2}
\end{equation}
and
\begin{equation}
F_{l+1/2,i+1/2}^{ee}=\frac{1}{v_{i+1/2}^{2}}\overline{n}_{e,l+1/2}
\end{equation}
while
\begin{equation}
B_{t,l+1/2,i+1/2}^{ee}=\frac{1}{2v_{i+1/2}}\overline{n}_{e,l+1/2}-\frac{1}{%
2v_{i+1/2}^{3}}\overline{n}_{e,l+1/2}\overline{T}_{e,l+1/2}
\end{equation}

Since no integrals appear in the coefficients, the expressions at momentum
flux grid points $i$ and $i+1$ can be obtained in a straightforward manner,
by just replacing $i+1/2$ by the corresponding index values. In that limit,
the first-order Legendre correction is neglected. In the calculations, both
electron-electron and electron-ion collision model in the high velocity
limit are used for a consistent description of the collisions.

\paragraph{Non-relativistic Lorentz model}

This very simple case corresponds to
\begin{equation}
A_{l+1/2,i+1/2}^{ee}=F_{l+1/2,i+1/2}^{ee}=B_{t,l+1/2,i+1/2}^{ee}=0
\end{equation}
and also on the momentum flux grid points $i$ and $i+1.$ Obviously, the
first-order Legendre correction is also neglected in that case.

\subsubsection{Electron-ion collision operator}

\paragraph{Non-relativistic Maxwellian background}

In that case matrix coefficients are

\begin{eqnarray}
A_{l+1/2,i+1/2}^{ei} &=&\frac{1}{2v_{i+1/2}}\sum_{s}\sum_{s^{\prime }}\frac{1%
}{u_{l+1/2,i+1/2}^{ss^{\prime }2}}\left[ erf \left(
u_{l+1/2,i+1/2}^{ss^{\prime }}\right) \right.  \nonumber \\
&&\left. -u_{l+1/2,i+1/2}erf ^{\prime }\left( u_{l+1/2,i+1/2}^{ss^{\prime
}}\right) \right] Z_{ss^{\prime }}^{2}\overline{n}_{ss^{\prime },l+1/2}
\end{eqnarray}

\begin{align}
&F_{l+1/2,i+1/2}^{ei}=\frac{1}{v_{i+1/2}^{2}}\sum_{s}\sum_{s^{\prime
}}\left[ erf \left( u_{l+1/2,i+1/2}^{ss^{\prime
}}\right)\right.\nonumber\\&\qquad\left.
-u_{l+1/2,i+1/2}^{ss^{\prime }}erf ^{\prime }\left(
u_{l+1/2,i+1/2}^{ss^{\prime }}\right) \right] Z_{ss^{\prime }}^{2}\frac{%
\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}_{s}}
\end{align}
and

\begin{eqnarray}
B_{t,l+1/2,i+1/2}^{ei} &=&\frac{1}{4v_{i+1/2}}\sum_{s}\sum_{s^{\prime }}%
\frac{1}{u_{l+1/2,i+1/2}^{ss^{\prime }2}}\left[ \left(
2u_{l+1/2,i+1/2}^{ss^{\prime }2}-1\right) erf \left(
u_{l+1/2,i+1/2}^{ss^{\prime }}\right) \right.  \nonumber \\
&&\left. +u_{l+1/2,i+1/2}^{ss^{\prime }}erf ^{\prime }\left(
u_{l+1/2,i+1/2}^{ss^{\prime }}\right) \right] Z_{ss^{\prime }}^{2}\overline{n%
}_{ss^{\prime },l+1/2}
\end{eqnarray}
where
\[
u_{l+1/2,i+1/2}^{ss^{\prime }}=\frac{v_{i+1/2}}{2v_{th,ss^{\prime
},l+1/2}^{\dagger }}=\frac{v_{i+1/2}}{2\sqrt{\overline{T}_{ss^{\prime
},l+1/2}/\overline{m}_{s}}}
\]

In the case $\overline{T}_{ss^{\prime },l+1/2}=0,$ to avoid a singularity,
\begin{equation}
A_{l+1/2,i+1/2}^{ei}=B_{t,l+1/2,i+1/2}^{ei}=0
\end{equation}
while
\begin{equation}
F_{l+1/2,i+1/2}^{ei}=\frac{1}{v_{i+1/2}^{2}}\sum_{s}\sum_{s^{\prime
}}Z_{ss^{\prime }}^{2}\frac{\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}%
_{s}}
\end{equation}
since $erf \left( +\infty \right) =1$ and $erf ^{\prime }\left( +\infty
\right) =0.$

Finally, for values of the matrix coefficients $A^{ei},F^{ei}$ and $%
B_{t}^{ei}$ on the momentum flux grids, one just has to replace $i+1/2$ by $%
i $ or $i+1$.

\paragraph{High-velocity limit}

In this limit, matrix coefficients are

\begin{equation}
A_{l+1/2,i+1/2}^{ei}=\frac{1}{v_{i+1/2}^{3}}\sum_{s}\sum_{s^{\prime
}}Z_{ss^{\prime }}^{2}\frac{\overline{n}_{ss^{\prime },l+1/2}\overline{T}%
_{ss^{\prime },l+1/2}}{\overline{m}_{s}}
\end{equation}
\begin{equation}
F_{l+1/2,i+1/2}^{ei}=\frac{1}{v_{i+1/2}^{2}}\sum_{s}\sum_{s^{\prime
}}Z_{ss^{\prime }}^{2}\frac{\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}%
_{s}}
\end{equation}
The double sum $\sum_{s}\sum_{s^{\prime }}$ takes into account of all ions
species $s$ in ionization state $s^{\prime }.$ Here, $\overline{n}%
_{ss^{\prime },l+1/2}$ is the normalized ion density at $\psi _{l+1/2},$ as
introduced in Sec. \ref{normtene}, and $\overline{m}_{s}$ is the ion rest
mass normalized to the electron rest mass $m_{e}$.

Coefficients $A^{ei}$ and $F^{ei}$ must be also calculated on the flux grids
and therefore, according to the previous definition,

\begin{equation}
A_{l+1/2,i}^{ei}=\frac{1}{v_{i}^{3}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime
}}^{2}\frac{\overline{n}_{ss^{\prime },l+1/2}\overline{T}_{ss^{\prime
},l+1/2}}{\overline{m}_{s}}
\end{equation}
and
\begin{equation}
F_{l+1/2,i}^{ei}=\frac{1}{v_{i}^{2}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime
}}^{2}\frac{\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}_{s}}
\end{equation}
for the grid points $i,$ while for the grid points $i+1,$ one has just to
replace $i$ by $i+1$ in the set of above expressions.

Furthermore, the expression of coefficient $B_{t}^{ei}$ is
\begin{equation}
B_{t,l+1/2,i+1/2}^{ei}=\frac{1}{2v_{i+1/2}}\sum_{s}\sum_{s^{\prime
}}Z_{ss^{\prime }}^{2}\overline{n}_{ss^{\prime },l+1/2}\left( 1-\frac{1}{%
v_{i+1/2}^{2}}\frac{\overline{T}_{ss^{\prime },l+1/2}}{\overline{m}_{s}}%
\right)
\end{equation}

\paragraph{Non-relativistic Lorentz model}

In that limit
\begin{equation}
A_{l+1/2,i+1/2}^{ei}=F_{l+1/2,i+1/2}^{ei}=0
\end{equation}
while

\begin{equation}
B_{t,l+1/2,i+1/2}^{ei}=1/2
\end{equation}

A straightforward extrapolation may be done for the momentum flux grid $i$
by $i+1.$

\subsection{Ohmic electric field}

According to the bounce averaged expression,
\begin{equation}
\overline{\overline{D}}_{p}^{E\left( 0\right) }\rightarrow \left\{
\begin{array}{l}
D_{pp,l+1/2,i+1,j+1/2}^{E\left( 0\right) }=0 \\
D_{pp,l+1/2,i,j+1/2}^{E\left( 0\right) }=0 \\
D_{p\xi ,l+1/2,i+1,j+1/2}^{E\left( 0\right) }=0 \\
D_{\xi p,l+1/2,i+1/2,j+1}^{E\left( 0\right) }=0 \\
D_{p\xi ,l+1/2,i+1/2,j+1/2}^{E\left( 0\right) }=0 \\
D_{\xi p,l+1/2,i+1/2,j+1/2}^{E\left( 0\right) }=0 \\
D_{p\xi ,l+1/2,i,j+1/2}^{E\left( 0\right) }=0 \\
D_{\xi p,l+1/2,i+1/2,j}^{E\left( 0\right) }=0 \\
D_{\xi \xi ,l+1/2,i+1/2,j+1}^{E\left( 0\right) }=0 \\
D_{\xi \xi ,l+1/2,i+1/2,j}^{E\left( 0\right) }=0
\end{array}
\right.
\end{equation}
and
\begin{equation}
\overline{\overline{F}}_{p}^{E\left( 0\right) }\rightarrow \left\{
\begin{array}{l}
F_{p,l+1/2,i+1,j+1/2}^{E\left( 0\right) }=\left( \overline{\lambda }%
_{1,-1,2}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) \xi _{0,j+1/2}%
\overline{E}_{\Vert 0,l+1/2} \\
F_{p,l+1/2,i,j+1/2}^{E\left( 0\right) }=\left( \overline{\lambda }%
_{1,-1,2}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) \xi _{0,j+1/2}%
\overline{E}_{\Vert 0,l+1/2} \\
F_{\xi ,l+1/2,i+1/2,j+1}^{E\left( 0\right) }=-\left( \overline{\lambda }%
_{1,-1,2}^{l+1/2,j+1}/\lambda ^{l+1/2,j+1}\right) \sqrt{1-\xi _{0,j+1}^{2}}%
\overline{E}_{\Vert 0,l+1/2} \\
F_{\xi ,l+1/2,i+1/2,j}^{E\left( 0\right) }=-\left( \overline{\lambda }%
_{1,-1,2}^{l+1/2,j}/\lambda ^{l+1/2,j}\right) \sqrt{1-\xi _{0,j}^{2}}%
\overline{E}_{\Vert 0,l+1/2}
\end{array}
\right.
\end{equation}
where $\overline{E}_{\Vert 0,l+1/2}$ is the parallel component of the Ohmic
electric field along the magnetic field line direction normalized to the
Dreicer field taken at the poloidal position where the magnetic field $%
\mathbf{B}$ is minimum, as explained in Sec.\ref{Efield}.

It is important to recall that the Ohmic electric field operator has a
cylindrical symmetry, while the description of the electron dynamics in
momentum space is based on the spherical symmetry of the collision operator.
As a consequence, there is a fundamental contradiction for the internal
boundary at $p=0$ where $p^{2}S_{p}=0$. Indeed, for large values of $%
\overline{E}_{\Vert 0},$ the maximum of the distribution function $%
f_{0}^{\left( 0\right) }$ is no more at $p=0$, but may be significantly
shifted along the axis $p_{\bot }=0$. Since in that extrem case $%
p^{2}S_{p}\neq 0$ at $p=0$ while it is naturally enforced to be null by
construction with the grid definition, the distribution function has a wrong
shape close to $p=0$ and the conservative scheme is no more preserved. It is
also important to note that the external boundary $\partial f_{0}^{\left(
0\right) }/\partial \xi _{0}=0$ at $\xi _{0}=0$ is also no more consistent
with initial bounce averaged assumptions, which represents also an important
failure of the use of the code.

Consequently, in order to avoid a misuse of the code, the range of validity
of $\overline{E}_{\Vert 0}$ is restricted so that condition
\begin{equation}
\left| \frac{F_{p}^{E\left( 0\right) }}{F_{p}^{C\left( 0\right) }}\right|
\ll 1
\end{equation}
is fullfiled at $p/p_{th}=1.$ Using the high-velicity limit of the collision
operator, this corresponds roughly to
\begin{equation}
\frac{\overline{E}_{\Vert 0}}{\overline{n}_{e}}\simeq \overline{E}_{\Vert
0}\ll 1
\end{equation}
since $\overline{n}_{e}\simeq 1$ in normalized units, as defined in Sec.\ref
{normtene}$.$ Consequently, flux surface averaged value of the Ohmic
electric field $\left\langle \overline{E}\right\rangle $ should be
restricted to $0.05$, in Dreicer units. A warning is indicated when this
value is exceeded.
