\paragraph{Belaiev-Budker model}

Here, coefficients are given for the Belaiev-Budker collision operator that
ranges from non-relativistic to fully relativistic regimes 
\begin{eqnarray}
A_{l+1/2,i+1/2}^{\left( 0\right) } &=&\frac{\left( F_{1,l+1/2,i+1/2}^{\left(
0\right) }+F_{2,l+1/2,i+1/2}^{\left( 0\right) }\right) }{v_{i+1/2}}\overline{%
T}_{e,l+1/2}  \nonumber \\
&&+\frac{1}{v_{i+1/2}^{3}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime }}^{2}%
\frac{\overline{n}_{ss^{\prime },l+1/2}\overline{T}_{ss^{\prime },l+1/2}}{%
\overline{m}_{s}}
\end{eqnarray}
and 
\begin{eqnarray}
F_{l+1/2,i+1/2}^{\left( 0\right) } &=&F_{1,l+1/2,i+1/2}^{\left( 0\right)
}+F_{2,l+1/2,i+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{1}{v_{i+1/2}^{2}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime }}^{2}%
\frac{\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}_{s}}
\end{eqnarray}
where 
\begin{equation}
F_{1,l+1/2,i+1/2}^{\left( 0\right) }=\frac{4\pi }{v_{i+1/2}^{2}}%
F_{11,l+1/2,i+1/2}^{\left( 0\right) }+\frac{4\pi }{p_{i+1/2}^{2}}%
F_{12,l+1/2,i+1/2}^{\left( 0\right) }
\end{equation}
and 
\begin{equation}
F_{2,l+1/2,i+1/2}^{\left( 0\right) }=\frac{4\pi }{v_{i+1/2}}\left( 1-\frac{%
\gamma _{i+1/2}\sigma _{i+1/2}}{z_{i+1/2}}\right) F_{21,l+1/2,i+1/2}^{\left(
0\right) }
\end{equation}

The sum $\sum_{ss^{\prime }}$ takes into account of all ions species $s$ in
ionization state $s^{\prime }.$ Here, $\overline{n}_{ss^{\prime },l+1/2}$ is
the normalized ion density at $\psi _{l+1/2},$ as introduced in Sec. ***,
and $\overline{m}_{s}$ is the ion rest mass normalized to the electron rest
mass $m_{e}$. Here, 
\begin{equation}
F_{11,l+1/2,i+1/2}^{\left( 0\right) }=\sum_{i^{\prime
}=0}^{_{i}}p_{i^{\prime }+1/2}v_{i^{\prime }+1/2}f_{M,l+1/2,i^{\prime
}+1/2}^{\left( 0\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
F_{12,l+1/2,i+1/2}^{\left( 0\right) }=\sum_{i^{\prime
}=0}^{_{i}}p_{i^{\prime }+1/2}v_{i^{\prime }+1/2}\left( 1-\frac{\gamma
_{i^{\prime }+1/2}\sigma _{i^{\prime }+1/2}}{z_{i^{\prime }+1/2}}\right)
f_{M,l+1/2,i^{\prime }+1/2}^{\left( 0\right) \left( k\right) }\Delta
p_{i^{\prime }+1/2}
\end{equation}
\begin{equation}
F_{21,l+1/2,i+1/2}^{\left( 0\right) }=\sum_{i^{\prime
}=i}^{_{n_{p}}}p_{i^{\prime }+1/2}f_{M,l+1/2,i^{\prime }+1/2}^{\left(
0\right) \left( k\right) }\Delta p_{i^{\prime }+1/2}
\end{equation}

Since coefficients $A^{\left( 0\right) },$ $B_{t}^{\left( 0\right) }$ and $%
F^{\left( 0\right) }$ are estimated on the flux momentum grid points $i$ and 
$i+1,$ an interpolation procedure must be carried out. Integrals are
approximated according to the rule 
\begin{eqnarray}
\int_{0}^{p_{i+1}}Xdp
&=&\int_{0}^{p_{i+1/2}}Xdp+\int_{p_{i+1/2}}^{p_{i+1}}Xdp  \nonumber \\
&\simeq &\int_{0}^{p_{i+1/2}}Xdp+X_{i+1/2}\left( p_{i+1}-p_{i+1/2}\right) 
\nonumber \\
&\simeq &\int_{0}^{p_{i+1/2}}Xdp+X_{i+1/2}\Delta p_{i+1/2}/2
\end{eqnarray}
since by definition, $p_{i+1/2}$ is placed exactly at the middle of the cell
between $p_{i}$ and $p_{i+1}.$ Much in the same way, 
\begin{eqnarray}
\int_{0}^{p_{i}}Xdp &=&\int_{0}^{p_{i+1/2}}Xdp-\int_{p_{i}}^{p_{i+1/2}}Xdp 
\nonumber \\
&\simeq &\int_{0}^{p_{i+1/2}}Xdp-X_{i+1/2}\Delta p_{i+1/2}/2
\end{eqnarray}

For integrals of the type 
\begin{eqnarray}
\int_{p_{i+1}}^{\infty }Xdp &=&\int_{p_{i+1/2}}^{\infty
}Xdp-\int_{p_{i+1/2}}^{p_{i+1}}Xdp  \nonumber \\
&\simeq &\int_{p_{i+1/2}}^{\infty }Xdp-X_{i+1/2}\Delta p_{i+1/2}/2
\end{eqnarray}
and 
\begin{eqnarray}
\int_{p_{i}}^{\infty }Xdp &=&\int_{p_{i+1/2}}^{\infty
}Xdp+\int_{p_{i}}^{p_{i+1/2}}Xdp  \nonumber \\
&\simeq &\int_{p_{i+1/2}}^{\infty }Xdp+X_{i+1/2}\Delta p_{i+1/2}/2
\end{eqnarray}

0\right) }+F_{2,l+1/2,i+1}^{\left( 0\right) }\right) }{v_{i}}\overline{T}%
_{e,l+1/2} \\
&&+\frac{1}{v_{i+1}^{3}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime }}^{2}\frac{%
\overline{n}_{ss^{\prime },l+1/2}\overline{T}_{ss^{\prime },l+1/2}}{%
\overline{m}_{s}}  \nonumber
\end{eqnarray}
and 
\begin{eqnarray}
F_{l+1/2,i+1}^{\left( 0\right) } &=&F_{1,l+1/2,i+1}^{\left( 0\right)
}+F_{2,l+1/2,i+1}^{\left( 0\right) } \\
&&+\frac{1}{v_{i+1}^{2}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime }}^{2}\frac{%
\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}_{s}}  \nonumber
\end{eqnarray}
where 
\begin{equation}
F_{1,l+1/2,i+1}^{\left( 0\right) }=\frac{4\pi }{v_{i+1}^{2}}%
F_{11,l+1/2,i+1}^{\left( 0\right) }+\frac{4\pi }{p_{i+1}^{2}}%

\paragraph{High velocity limit}

Though the high velocity limit corresponds to a restricted range of
application regarding the full collision operator, it can contribute to
useful comparison with some theoretical calculations. Therefore, this
possibility has been implemented in the code. In that case, expressions of
the coefficients are greatly simplified,

\begin{eqnarray}

_{e,l+1/2}\overline{T}_{e,l+1/2}  \nonumber \\
&&+\frac{1}{v_{i+1/2}^{3}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime }}^{2}%
\frac{\overline{n}_{ss^{\prime },l+1/2}\overline{T}_{ss^{\prime },l+1/2}}{%
\overline{m}_{s}}

&&+\frac{1}{v_{i+1/2}^{2}}\sum_{s}\sum_{s^{\prime }}Z_{ss^{\prime }}^{2}%
\frac{\overline{n}_{ss^{\prime },l+1/2}}{\overline{m}_{s}}
\end{eqnarray}
