
In this section, only the electron dynamics in configuration space is
discussed, at fixed $p_{i+1/2}$ and $\xi _{0,j+1/2},$

\begin{eqnarray}
&&\left. \frac{\widetilde{q}}{B_{\min }}p^{2}\frac{\partial f^{\left(
0\right) }}{\partial \psi }\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}  \nonumber \\
&=&\frac{p_{i+1/2}^{2}}{\lambda ^{l+1/2,j+1/2}}\times  \nonumber \\
&&\left. \frac{\partial }{\partial \psi }\left( R_{\min }\widetilde{q}\frac{%
B_{P\min }}{B_{\min }}\lambda \mathbf{S}_{\psi }^{\left( 0\right) }\right)
\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{eqnarray}
where 
\begin{equation}
\mathbf{S}_{\psi }^{\left( 0\right) }=-D_{\psi \psi }^{\left( 0\right)
}\left| \nabla \psi \right| _{\min }\frac{\partial f_{0}^{\left( 0\right) }}{%
\partial \psi }+F_{\psi }^{\left( 0\right) }f_{0}^{\left( 0\right) }
\end{equation}
since cross-terms between momentum and configuration spaces are neglected,
i.e. $D_{p\psi }^{\left( 0\right) }=D_{\psi p}^{\left( 0\right) }=D_{\xi
\psi }^{\left( 0\right) }=D_{\psi \xi }^{\left( 0\right) }=0.$

Therefore,

\begin{equation}
\left. \frac{\widetilde{q}}{B_{\min }}p^{2}\frac{\partial f^{\left( 0\right)
}}{\partial \psi }\right| _{l+1/2,i+1/2,j+1/2}^{(k+1)}=\frac{p_{i+1/2}^{2}}{%
\lambda ^{l+1/2,j+1/2}}\sum_{m=1}^{m=4}T^{\left[ m\right] }
\end{equation}
with 
\begin{eqnarray}
T^{\left[ 1\right] } &=&-D_{\psi \psi ,l+1,i+1/2,j+1/2}^{\left( 0\right)
}\left[ R_{\min ,l+1}^{2}\widetilde{q}_{l+1}\frac{B_{P\min ,l+1}^{2}}{%
B_{\min ,l+1}}\lambda ^{l+1,j+1/2}\right] \times   \nonumber \\
&&\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \psi }\right|
_{l+1,i+1/2,j+1/2}^{\left( k+1\right) }/\Delta \psi _{l+1/2}
\end{eqnarray}
\begin{eqnarray}
T^{\left[ 2\right] } &=&F_{\psi ,l+1,i+1/2,j+1/2}^{\left( 0\right) }\left[
R_{\min ,l+1}\widetilde{q}_{l+1}\frac{B_{P\min ,l+1}}{B_{\min ,l+1}}\lambda
^{l+1,j+1/2}\right] \times   \nonumber \\
&&f_{0,l+1,i+1/2,j+1/2}^{\left( 0\right) \left( k+1\right) }/\Delta \psi
_{l+1/2}
\end{eqnarray}
\begin{eqnarray}
T^{\left[ 3\right] } &=&+D_{\psi \psi ,l,i+1/2,j+1/2}^{\left( 0\right)
}\left[ R_{\min ,l}^{2}\widetilde{q}_{l}\frac{B_{P\min ,l}^{2}}{B_{\min ,l}}%
\lambda ^{l,j+1/2}\right] \mathbf{\times }  \nonumber \\
&&\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \psi }\right|
_{l,i+1/2,j+1/2}^{\left( k+1\right) }/\Delta \psi _{l+1/2}
\end{eqnarray}
\begin{eqnarray}
T^{\left[ 4\right] } &=&-F_{\psi ,l,i+1/2,j+1/2}^{\left( 0\right) }\left[
R_{\min ,l}\widetilde{q}_{l}\frac{B_{P\min ,l}}{B_{\min ,l}}\lambda
^{l,j+1/2}\right] \mathbf{\times }  \nonumber \\
&&f_{0,l,i+1/2,j+1/2}^{\left( 0\right) \left( k+1\right) }/\Delta \psi
_{l+1/2}
\end{eqnarray}
and discrete expressions of the partial derivatives are, 
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \psi }\right|
_{l+1,i+1/2,j+1/2}^{(k+1)}=\frac{f_{0,l+3/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta \psi _{l+1}}
\end{equation}
\begin{equation}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \psi }\right|
_{l,i+1/2,j+1/2}^{(k+1)}=\frac{f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l-1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta \psi _{l}}
\end{equation}

As for the dynamics in momentum space, interpolation between full and half
grids must be performed, since some values of the distribution function $%
f_{0}^{\left( 0\right) }$ must be determined on the flux grid. Therefore 
\begin{eqnarray}
f_{0,l+1,i+1/2,j+1/2}^{\left( 0\right) (k+1)} &=&\left( 1-\delta _{\psi
,l+1,i+1/2,j+1/2}^{\left( 0\right) }\right) f_{0,l+3/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}  \nonumber \\
&&+\delta _{\psi ,l+1,i+1/2,j+1/2}^{\left( 0\right)
}f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{eqnarray}
\begin{eqnarray}
f_{0,l,i+1/2,j+1/2}^{\left( 0\right) (k+1)} &=&\left( 1-\delta _{\psi
,l,i+1/2,j+1/2}^{\left( 0\right) }\right) f_{0,l+1/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}  \nonumber \\
&&+\delta _{\psi ,l,i+1/2,j+1/2}^{\left( 0\right)
}f_{0,l-1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{eqnarray}

Gathering all terms, matrix coefficients for the zero order Fokker-Planck
equation, according to the relation, 
\begin{eqnarray}
\left. \frac{\widetilde{q}\left( \psi \right) }{B_{\min }\left( \psi \right) 
}p^{2}\frac{\partial f^{\left( 0\right) }}{\partial \psi }\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)} &=&\frac{p_{i+1/2}^{2}}{\lambda ^{l+1/2,j+1/2}}%
\times  \nonumber \\
&&\sum_{l^{\prime }=l-1}^{l^{\prime }=l+1}\overline{\overline{M}}_{\psi
,l^{\prime }+1/2,i+1/2,j+1/2}^{\left( 0\right) }f_{0,l^{\prime
}+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{eqnarray}
are

\begin{eqnarray}
\overline{\overline{M}}_{\psi ,l+3/2,i+1/2,j+1/2}^{\left( 0\right) } &=&-%
\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}\Delta
\psi _{l+1}}R_{\min ,l+1}B_{P\min ,l+1}D_{\psi \psi
,l+1,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}}%
F_{\psi ,l+1,i+1/2,j+1/2}^{\left( 0\right) }\left( 1-\delta _{\psi
,l+1,i+1/2,j+1/2}^{\left( 0\right) }\right)
\end{eqnarray}
\begin{eqnarray}
\overline{\overline{M}}_{\psi ,l+1/2,i+1/2,j+1/2}^{\left( 0\right) } &=&+%
\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}\Delta
\psi _{l+1}}R_{\min ,l+1}B_{P\min ,l+1}D_{\psi \psi
,l+1,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}}
F_{\psi ,l+1,i+1/2,j+1/2}^{\left( 0\right) }\delta _{\psi
,l+1,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}\Delta
\psi _{l}}R_{\min ,l}B_{P\min ,l}D_{\psi \psi ,l,i+1/2,j+1/2}^{\left(
0\right) }  \nonumber \\
&&-\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}}F_{\psi
,l,i+1/2,j+1/2}^{\left( 0\right) }\left( 1-\delta _{\psi
,l,i+1/2,j+1/2}^{\left( 0\right) }\right)
\end{eqnarray}
\begin{eqnarray}
\overline{\overline{M}}_{\psi ,l-1/2,i+1/2,j+1/2}^{\left( 0\right) } &=&-%
\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}\Delta \psi
_{l}}R_{\min ,l}B_{P\min ,l}D_{\psi \psi ,l,i+1/2,j+1/2}^{\left( 0\right) } 
\nonumber \\
&&-\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \psi _{l+1/2}}F_{\psi
,l,i+1/2,j+1/2}^{\left( 0\right) }\delta _{\psi ,l,i+1/2,j+1/2}^{\left(
0\right) }
\end{eqnarray}
with 
\begin{equation}
\alpha _{l,j+1/2}^{\left( 0\right) }=R_{\min ,l}\widetilde{q}_{l}\frac{%
B_{P\min ,l}}{B_{\min ,l}}\lambda ^{l,j+1/2}
\end{equation}
\begin{equation}
\alpha _{l+1,j+1/2}^{\left( 0\right) }=R_{\min ,l+1}\widetilde{q}_{l+1}\frac{%
B_{P\min ,l+1}}{B_{\min ,l+1}}\lambda ^{l+1,j+1/2}
\end{equation}

As expected, spatial transport corresponds to a simple tri-diagonal matrix.
