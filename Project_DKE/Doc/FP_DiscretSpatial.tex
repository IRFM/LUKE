
In this section, only the electron dynamics in configuration space is
discussed, at fixed $p_{i+1/2}$ and $\xi _{0,j+1/2},$  

\begin{eqnarray}
&&\left. \frac{\widetilde{q}\left( \rho \right) }{B_{\min }\left( \rho
\right) }p^{2}\frac{\partial f^{\left( 0\right) }}{\partial \rho }\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}  \nonumber \\
&=&p_{i+1/2}^{2}\frac{\rho _{\psi ,_{l+1/2}}^{\prime }}{\lambda \left( \rho
_{l+1/2},\xi _{0,j+1/2}\right) }\left. \frac{\partial }{\partial \rho }%
\left( R_{\min }\left( \rho \right) \widetilde{q}\left( \rho \right) \frac{%
B_{P\min }\left( \rho \right) }{B_{\min }\left( \rho \right) }\lambda \left(
\rho ,\xi _{0}\right) \mathbf{S}_{\rho }^{\left( 0\right) }\right) \right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}
\end{eqnarray}
where
\begin{equation}
\mathbf{S}_{\rho }^{\left( 0\right) }=-D_{\rho \rho }^{\left( 0\right) }\rho
_{\psi }^{\prime }\frac{\partial f_{0}^{\left( 0\right) }}{\partial \rho }%
+F_{\rho }^{\left( 0\right) }f_{0}^{\left( 0\right) }
\end{equation}
since cross-terms between momentum and configuration spaces are neglected,
i.e. $D_{p\psi }^{\left( 0\right) }=D_{\psi p}^{\left( 0\right) }=D_{\xi
\psi }^{\left( 0\right) }=D_{\psi \xi }^{\left( 0\right) }=0.$ 

Therefore,

\begin{eqnarray}
\left. \frac{\widetilde{q}\left( \rho \right) }{B_{\min }\left( \rho \right) 
}p^{2}\frac{\partial f^{\left( 0\right) }}{\partial \rho }\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)} &=&p_{i+1/2}^{2}\frac{\rho _{\psi
,_{l+1/2}}^{\prime }}{\lambda \left( \rho _{l+1/2},\xi _{0,j+1/2}\right) }%
\left[ R_{\min }\left( \rho _{l+1}\right) \widetilde{q}\left( \rho
_{l+1}\right) \frac{B_{P\min }\left( \rho _{l+1}\right) }{B_{\min }\left(
\rho _{l+1}\right) }\lambda \left( \rho _{l+1},\xi _{0,j+1/2}\right) \right] 
\mathbf{\times }  \nonumber \\
&&\left[ -D_{\rho \rho ,l+1,i+1/2,j+1/2}^{\left( 0\right) }\rho _{\psi
,l+1}^{\prime }\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \rho 
}\right| _{l+1,i+1/2,j+1/2}^{\left( k+1\right) }+F_{\rho
,l+1,i+1/2,j+1/2}^{\left( 0\right) }f_{0,l+1,i+1/2,j+1/2}^{\left( 0\right)
\left( k+1\right) }\right] /\Delta \rho _{l+1/2}  \nonumber \\
&&-p_{i+1/2}^{2}\frac{\rho _{\psi ,_{l+1/2}}^{\prime }}{\lambda \left( \rho
_{l+1/2},\xi _{0,j+1/2}\right) }\left[ R_{\min }\left( \rho _{l}\right) 
\widetilde{q}\left( \rho _{l}\right) \frac{B_{P\min }\left( \rho _{l}\right) 
}{B_{\min }\left( \rho _{l}\right) }\lambda \left( \rho _{l},\xi
_{0,j+1/2}\right) \right] \mathbf{\times }  \nonumber \\
&&\left[ -D_{\rho \rho ,l,i+1/2,j+1/2}^{\left( 0\right) }\rho _{\psi
,l}^{\prime }\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \rho }%
\right| _{l,i+1/2,j+1/2}^{\left( k+1\right) }+F_{\rho
,l,i+1/2,j+1/2}^{\left( 0\right) }f_{0,l,i+1/2,j+1/2}^{\left( 0\right)
\left( k+1\right) }\right] /\Delta \rho _{l+1/2}
\end{eqnarray}
and discrete expressions of the partial derivatives are summarized
\begin{equation}
\left\{ 
\begin{array}{l}
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \rho }\right|
_{l+1,i+1/2,j+1/2}^{(k+1)}=\frac{f_{0,l+3/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta \rho _{l+1}}
\\ 
\left. \frac{\partial f_{0}^{\left( 0\right) }}{\partial \rho }\right|
_{l,i+1/2,j+1/2}^{(k+1)}=\frac{f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right)
(k+1)}-f_{0,l-1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}}{\Delta \rho _{l}}
\end{array}
\right. 
\end{equation}

As for the dynamics in momentum space, interpolation between full and half
grids must be performed, when the value of the distribution function $%
f_{0}^{\left( 0\right) }$ must be detrmined on the flux grid. Hence
\begin{equation}
\left\{ 
\begin{array}{l}
f_{0,l+1,i+1/2,j+1/2}^{\left( 0\right) (k+1)}=\left( 1-\delta _{\rho
,l+1,i+1/2,j+1/2}^{\left( 0\right) }\right) f_{0,l+3/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}+\delta _{\rho ,l+1,i+1/2,j+1/2}^{\left( 0\right)
}f_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)} \\ 
f_{0,l,i+1/2,j+1/2}^{\left( 0\right) (k+1)}=\left( 1-\delta _{\rho
,l,i+1/2,j+1/2}^{\left( 0\right) }\right) f_{0,l+1/2,i+1/2,j+1/2}^{\left(
0\right) (k+1)}+\delta _{\rho ,l,i+1/2,j+1/2}^{\left( 0\right)
}f_{0,l-1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{array}
\right. 
\end{equation}

Gathering all terms matrix coefficients for the zero order Fokker-Planck
equation according to the relation,
\begin{equation}
\left. \frac{\widetilde{q}\left( \rho \right) }{B_{\min }\left( \rho \right) 
}p^{2}\frac{\partial f^{\left( 0\right) }}{\partial \rho }\right|
_{l+1/2,i+1/2,j+1/2}^{(k+1)}=p_{i+1/2}^{2}\frac{\rho _{\psi
,_{l+1/2}}^{\prime }}{\lambda \left( \rho _{l+1/2},\xi _{0,j+1/2}\right) }%
\sum_{l^{\prime }=l-1}^{l^{\prime }=l+1}\overline{\overline{M}}_{\rho
,l^{\prime }+1/2,i+1/2,j+1/2}^{\left( 0\right) }f_{0,l^{\prime
}+1/2,i+1/2,j+1/2}^{\left( 0\right) (k+1)}
\end{equation}
are

\begin{eqnarray}
\overline{\overline{M}}_{\rho ,l+3/2,i+1/2,j+1/2}^{\left( 0\right) } &=&-%
\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}\Delta %
\rho _{l+1}}D_{\rho \rho ,l+1,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}}%
F_{\rho ,l+1,i+1/2,j+1/2}^{\left( 0\right) }\left( 1-\delta _{\rho
,l+1,i+1/2,j+1/2}^{\left( 0\right) }\right) 
\end{eqnarray}
\begin{eqnarray}
\overline{\overline{M}}_{\rho ,l+1/2,i+1/2,j+1/2}^{\left( 0\right) } &=&+%
\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}\Delta %
\rho _{l+1}}D_{\rho \rho ,l+1,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{\alpha _{l+1,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}}%
F_{\rho ,l+1,i+1/2,j+1/2}^{\left( 0\right) }\delta _{\rho
,l+1,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&+\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}\Delta %
\rho _{l}}D_{\rho \rho ,l,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&-\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}}F_{\rho
,l,i+1/2,j+1/2}^{\left( 0\right) }\left( 1-\delta _{\rho
,l,i+1/2,j+1/2}^{\left( 0\right) }\right) 
\end{eqnarray}
\begin{eqnarray}
\overline{\overline{M}}_{\rho ,l-1/2,i+1/2,j+1/2}^{\left( 0\right) } &=&-%
\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}\Delta \rho
_{l}}D_{\rho \rho ,l,i+1/2,j+1/2}^{\left( 0\right) }  \nonumber \\
&&-\frac{\alpha _{l,j+1/2}^{\left( 0\right) }}{\Delta \rho _{l+1/2}}F_{\rho
,l,i+1/2,j+1/2}^{\left( 0\right) }\delta _{\rho ,l,i+1/2,j+1/2}^{\left(
0\right) }
\end{eqnarray}
with
\begin{equation}
\left\{ 
\begin{array}{l}
\alpha _{l,j+1/2}^{\left( 0\right) }=R_{\min }\left( \rho _{l}\right) 
\widetilde{q}\left( \rho _{l}\right) \frac{B_{P\min }\left( \rho _{l}\right) 
}{B_{\min }\left( \rho _{l}\right) }\lambda \left( \rho _{l},\xi
_{0,j+1/2}\right)  \\ 
\alpha _{l+1,j+1/2}^{\left( 0\right) }=R_{\min }\left( \rho _{l+1}\right) 
\widetilde{q}\left( \rho _{l+1}\right) \frac{B_{P\min }\left( \rho
_{l+1}\right) }{B_{\min }\left( \rho _{l+1}\right) }\lambda \left( \rho
_{l+1},\xi _{0,j+1/2}\right) 
\end{array}
\right. 
\end{equation}

As expected, spatial transport corresponds to a simple tri-diagonal matrix.
