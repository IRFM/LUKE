
\paragraph{ Momentum space operator}

It is possible to recover the general bounce averaged transport equation in
momentum space by another independent approach. Here, the momentum space
dynamics of the kinetic equation is be expressed in conservative form as (%
\ref{consform}) 
\begin{equation}
\left. \frac{\partial f}{\partial t}\right| _{\mathbf{p}}=\nabla _{\mathbf{p}
}\cdot \mathbf{S}_{\mathbf{p}}
\end{equation}
which expands as 
\begin{equation}
\left. \frac{\partial f}{\partial t}\right| _{\mathbf{p}}=\frac{1}{J_{%
\mathbf{p}}}\frac{\partial }{\partial p^{i}}\left( J_{\mathbf{p}}S_{\mathbf{p%
}}^{i}\right)
\end{equation}
where $J_{\mathbf{p}}$ is the momentum space Jacobian , and using the
momentum space coordinate system is $\left( p,\xi ,\varphi \right) $,
described in (\ref{pxidef}). The spherical system has the natural symmetry
of collisions. The momentum space Jacobian is (\ref{jacpxi}) 
\begin{equation}
J_{\mathbf{p}}=p^{2}
\end{equation}
so that, taking into account that the equation was gyroaveraged and
therefore the coordinate $\varphi $ disappears, we get the following
expression for the divergence (\ref{pxidiv}) 
\begin{equation}
\left. \frac{\partial f_{0}}{\partial t}\right| _{\mathbf{p}}=\frac{1}{p^{2}}%
\frac{\partial }{\partial p}\left( p^{2}S_{p}\right) -\frac{1}{p}\frac{%
\partial }{\partial \xi }\left( \sqrt{1-\xi ^{2}}S_{\xi }\right)
\end{equation}
where we define 
\begin{eqnarray}
S_{p} &=&\mathbf{S}_{\mathbf{p}}\cdot \widehat{p} \\
S_{\xi } &=&\mathbf{S}_{\mathbf{p}}\cdot \widehat{\xi }  \nonumber
\end{eqnarray}

We can generally decompose the momentum space fluxes into diffusive and
convective parts (\ref{momflux}) 
\begin{equation}
\mathbf{S}_{\mathbf{p}}=-\Bbb{D}_{\mathbf{p}}\nabla _{\mathbf{p}}f+\mathbf{F}%
_{\mathbf{p}}f  \label{momflux2}
\end{equation}
with 
\begin{eqnarray}
\Bbb{D}_{\mathbf{p}} &=&\left( 
\begin{array}{ll}
D_{pp} & D_{p\xi } \\ 
D_{\xi p} & D_{\xi \xi }
\end{array}
\right) \\
\mathbf{F}_{\mathbf{p}} &=&\left( 
\begin{array}{l}
F_{p} \\ 
F_{\xi }
\end{array}
\right)
\end{eqnarray}

The gradient in $\left( p,\xi \right) $ coordinates in given by (\ref
{pxigrad}) 
\begin{equation}
\nabla _{\mathbf{p}}=\left( 
\begin{array}{l}
\nabla _{p} \\ 
\nabla _{\xi }
\end{array}
\right)
\end{equation}
with 
\begin{eqnarray}
\nabla _{p} &=&\frac{\partial }{\partial p} \\
\nabla _{\xi } &=&-\frac{\sqrt{1-\xi ^{2}}}{p}\frac{\partial }{\partial \xi }
\end{eqnarray}
so that we finally get 
\begin{eqnarray}
S_{p} &=&-D_{pp}\frac{\partial f}{\partial p}+\frac{\sqrt{1-\xi ^{2}}}{p}%
D_{p\xi }\frac{\partial f}{\partial \xi }+F_{p}f  \label{momflux3} \\
S_{\xi } &=&-D_{\xi p}\frac{\partial f}{\partial p}+\frac{\sqrt{1-\xi ^{2}}}{%
p}D_{\xi \xi }\frac{\partial f}{\partial \xi }+F_{\xi }f  \nonumber
\end{eqnarray}

\paragraph{Bounce-averaged operator}

The bounce averaged operator is 
\begin{equation}
\left\{ \left. \frac{\partial f}{\partial t}\right| _{\mathbf{p}}\right\}
=\left\{ \frac{1}{p^{2}}\frac{\partial }{\partial p}\left( p^{2}S_{p}\right)
\right\} -\left\{ \frac{1}{p}\frac{\partial }{\partial \xi }\left( \sqrt{%
1-\xi ^{2}}S_{\xi }\right) \right\}  \label{kinbav}
\end{equation}
where the bounce averaging operation was defined in (\ref{bav}) 
\begin{equation}
\left\{ \mathcal{A}\right\} =\frac{1}{\lambda \widetilde{q}}\left[ \frac{1}{2%
}\sum_{\sigma }\right] _{T}\int_{\theta _{\min }}^{\theta _{\max }}\frac{%
d\theta }{2\pi }\frac{1}{\left| \widehat{\psi }\cdot \widehat{r}\right| }%
\frac{r}{R_{0}}\frac{B}{B_{P}}\frac{\xi _{0}}{\xi }\mathcal{A}
\end{equation}
and $\xi $ is given along the trajectory by 
\begin{equation}
\xi \left( \psi ,\theta ,\xi _{0}\right) =\sigma \sqrt{1-\Psi \left( \psi
,\theta \right) \left( 1-\xi _{0}^{2}\right) }  \label{xi2}
\end{equation}
with 
\begin{equation}
\Psi \left( \psi ,\theta \right) =\frac{B\left( \psi ,\theta \right) }{
B_{\min }\left( \psi \right) }
\end{equation}

We find from (\ref{xi2}) that in momentum space 
\begin{equation}
\xi d\xi =\Psi \xi _{0}d\xi _{0}
\end{equation}
and we also get 
\begin{equation}
\left( 1-\xi ^{2}\right) =\Psi \left( 1-\xi _{0}^{2}\right)
\end{equation}

Then, keeping in mind that $\left| \xi _{0}\right| =\sigma \xi _{0}$ is
independent of $\sigma $, we can transform as follows, 
\begin{eqnarray}
\left\{ \frac{1}{p}\frac{\partial }{\partial \xi }\left( \sqrt{1-\xi ^{2}}%
S_{\xi }\right) \right\}  &=&\frac{1}{\lambda \widetilde{q}}\left[ \frac{1}{2%
}\sum_{\sigma }\right] _{T}\int_{\theta _{\min }}^{\theta _{\max }}\frac{%
d\theta }{2\pi }\frac{1}{\left| \widehat{\psi }\cdot \widehat{r}\right| }%
\frac{r}{R_{0}}\frac{B}{B_{P}}\frac{\xi _{0}}{\xi }\frac{1}{p}\frac{\partial 
}{\partial \xi }\left( \sqrt{1-\xi ^{2}}S_{\xi }\right)   \nonumber \\
&=&\frac{1}{\lambda p}\frac{\partial }{\partial \sigma \xi _{0}}\frac{1}{%
\widetilde{q}}\left[ \frac{1}{2}\sum_{\sigma }\right] _{T}\int_{\theta
_{\min }}^{\theta _{\max }}\frac{d\theta }{2\pi }\frac{1}{\left| \widehat{%
\psi }\cdot \widehat{r}\right| }\frac{r}{R_{0}}\frac{B}{B_{P}}\frac{\sigma }{%
\Psi }\left( \sqrt{1-\xi ^{2}}S_{\xi }\right)   \nonumber \\
&=&\frac{1}{\lambda p}\frac{\partial }{\partial \xi _{0}}\sqrt{1-\xi _{0}^{2}%
}\sigma \frac{1}{\widetilde{q}}\left[ \frac{1}{2}\sum_{\sigma }\right]
_{T}\int_{\theta _{\min }}^{\theta _{\max }}\frac{d\theta }{2\pi }\frac{1}{%
\left| \widehat{\psi }\cdot \widehat{r}\right| }\frac{r}{R_{0}}\frac{B}{B_{P}%
}\frac{\sigma }{\sqrt{\Psi }}\left( S_{\xi }\right)   \nonumber \\
&=&\frac{1}{\lambda p}\frac{\partial }{\partial \xi _{0}}\sqrt{1-\xi _{0}^{2}%
}\lambda \sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}S_{\xi
}\right\} 
\end{eqnarray}

Finally, we can rewrite (\ref{kinbav}) under a conservative form as 
\begin{equation}
\left\{ \left. \frac{\partial f}{\partial t}\right| _{\mathbf{p}}\right\} =%
\frac{1}{p^{2}}\frac{\partial }{\partial p}\left( p^{2}S_{p}^{\left(
0\right) }\right) -\frac{1}{\lambda p}\frac{\partial }{\partial \xi _{0}}%
\left( \sqrt{1-\xi _{0}^{2}}\lambda S_{\xi _{0}}^{\left( 0\right) }\right)
\label{kinbav2}
\end{equation}
where the following components are defined 
\begin{eqnarray}
S_{p}^{\left( 0\right) } &=&\left\{ S_{p}\right\}  \label{sp0} \\
S_{\xi _{0}}^{\left( 0\right) } &=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{%
\Psi }\xi _{0}}S_{\xi }\right\}  \nonumber
\end{eqnarray}

Here, expression (\ref{kinbav2}) is completely equivalent to the momentum
transport equation deduced from particle conservation. However, this
equivalence may be used only because the bounce-averaged operator is local,
and does not depends of $\psi .$
