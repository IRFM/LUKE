The Momentum Dynamics of the Kinetic equation can be expressed in
conservative form as
\begin{equation}
J_{{\bf p}}\left. \frac{\partial f_{0}}{\partial t}\right| _{{\bf p}}=J_{%
{\bf p}}\nabla _{{\bf p}}\cdot {\bf S}_{{\bf p}}
\end{equation}
where the multiplication by the momentum space Jacobian $J_{{\bf p}}$ is
introduced in order to avoid singularities. It expands as 
\begin{equation}
J_{{\bf p}}\left. \frac{\partial f_{0}}{\partial t}\right| _{{\bf p}}=J_{%
{\bf p}}\left[ \frac{1}{J_{{\bf p}}}\frac{\partial }{\partial p_{i}}\left(
J_{{\bf p}}S_{p_{i}}\right) \right] 
\end{equation}
and the momentum space Jacobian is
\begin{equation}
J_{{\bf p}}=p^{2}
\end{equation}
so that, taking into account that the equation was gyroaveraged, we get
\begin{equation}
J_{{\bf p}}\left. \frac{\partial f_{0}}{\partial t}\right| _{{\bf p}}=\frac{%
\partial }{\partial p}\left( p^{2}S_{p}\right) -p\frac{\partial }{\partial %
\xi }\left( \sqrt{1-\xi ^{2}}S_{\xi }\right) 
\end{equation}

The bounce averaged equation is
\begin{equation}
\left\{ J_{{\bf p}}\left. \frac{\partial f_{0}}{\partial t}\right| _{{\bf p}%
}\right\} =\left\{ \frac{\partial }{\partial p}\left( p^{2}S_{p}\right)
\right\} -\left\{ p\frac{\partial }{\partial \xi }\left( \sqrt{1-\xi ^{2}}%
S_{\xi }\right) \right\}   \label{linbav}
\end{equation}
where the bounce averaging operation was defined in (\ref{bav})
\begin{equation}
\left\{ {\cal A}\right\} =\frac{1}{\lambda \widetilde{q}}\left[ \frac{1}{2}%
\sum_{\sigma }\right] _{T}\int_{\theta _{\min }}^{\theta _{\max }}\frac{%
d\theta }{2\pi }\frac{1}{\left| \widehat{\psi }\cdot \widehat{r}\right| }%
\frac{r}{R_{0}}\frac{B}{B_{P}}\frac{\xi _{0}}{\xi }{\cal A}
\end{equation}
and $\xi $ is given along the trajectory by
\begin{equation}
\xi \left( \psi ,\theta ,\xi _{0}\right) =\sigma \sqrt{1-\Psi \left( \psi
,\theta \right) \left( 1-\xi _{0}^{2}\right) }  \label{xi2}
\end{equation}
with
\begin{equation}
\Psi \left( \psi ,\theta \right) =\frac{B\left( \psi ,\theta \right) }{%
B_{\min }\left( \psi \right) }
\end{equation}

We find from (\ref{xi2}) that in momentum space
\begin{equation}
\xi d\xi =\Psi \xi _{0}d\xi _{0}
\end{equation}
and we also get
\begin{equation}
\left( 1-\xi ^{2}\right) =\Psi \left( 1-\xi _{0}^{2}\right) 
\end{equation}

Then, keeping in mind that $\left| \xi _{0}\right| =\sigma \xi _{0}$ is
independent of $\sigma $, we can transform as follows, 
\begin{eqnarray}
\left\{ p\frac{\partial }{\partial \xi }\left( \sqrt{1-\xi ^{2}}S_{\xi
}\right) \right\}  &=&\frac{1}{\lambda \widetilde{q}}\left[ \frac{1}{2}%
\sum_{\sigma }\right] _{T}\int_{\theta _{\min }}^{\theta _{\max }}\frac{%
d\theta }{2\pi }\frac{rB/\left( R_{0}B_{P}\right) }{\left| \widehat{\psi }%
\cdot \widehat{r}\right| }\frac{\xi _{0}}{\xi }p\frac{\partial }{\partial %
\xi }\left( \sqrt{1-\xi ^{2}}S_{\xi }\right)  \\
&=&\frac{p}{\lambda }\frac{\partial }{\partial \sigma \xi _{0}}\frac{1}{%
\widetilde{q}}\left[ \frac{1}{2}\sum_{\sigma }\right] _{T}\int_{\theta
_{\min }}^{\theta _{\max }}\frac{d\theta }{2\pi }\frac{rB/\left(
R_{0}B_{P}\right) }{\left| \widehat{\psi }\cdot \widehat{r}\right| }\frac{%
\sigma }{\Psi }\left( \sqrt{1-\xi ^{2}}S_{\xi }\right)  \\
&=&\frac{p}{\lambda }\frac{\partial }{\partial \xi _{0}}\sqrt{1-\xi _{0}^{2}}%
\sigma \frac{1}{\lambda \widetilde{q}}\left[ \frac{1}{2}\sum_{\sigma
}\right] _{T}\int_{\theta _{\min }}^{\theta _{\max }}\frac{d\theta }{2\pi }%
\frac{rB/\left( R_{0}B_{P}\right) }{\left| \widehat{\psi }\cdot \widehat{r}%
\right| }\frac{\sigma }{\sqrt{\Psi }}\left( S_{\xi }\right)  \\
&=&\frac{p}{\lambda }\frac{\partial }{\partial \xi _{0}}\sqrt{1-\xi _{0}^{2}}%
\lambda \sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}S_{\xi
}\right\} 
\end{eqnarray}
Finally, we can rewrite (\ref{kinbav}) under a conservative form as
\begin{equation}
\left\{ J_{{\bf p}}\left. \frac{\partial f_{0}}{\partial t}\right| _{{\bf p}%
}\right\} =\frac{\partial }{\partial p}\left( p^{2}S_{p}^{\left( 0\right)
}\right) -\frac{p}{\lambda }\frac{\partial }{\partial \xi _{0}}\left( \sqrt{%
1-\xi _{0}^{2}}\lambda S_{\xi _{0}}^{\left( 0\right) }\right) 
\label{kinbav2}
\end{equation}
with 
\begin{eqnarray}
S_{p}^{\left( 0\right) } &=&\left\{ S_{p}\right\}   \label{sp0} \\
S_{\xi _{0}}^{\left( 0\right) } &=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{%
\Psi }\xi _{0}}S_{\xi }\right\} 
\end{eqnarray}
