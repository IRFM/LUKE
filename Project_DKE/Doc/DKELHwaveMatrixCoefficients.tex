On a \[
D^{\text{LH}}(p,\xi)=\frac{p_{\Vert}^{2}}{p_{\bot}^{2}}\frac{v_{Te}}{\left|v_{\Vert}\right|}H(v_{\Vert}-v_{\min})H(v_{\max}-v_{\Vert})2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\]
 avec \[
D_{0}^{\text{LH}}=\lim_{V\rightarrow\infty}\frac{\pi q_{e}^{2}\Delta\theta}{V}\int\frac{d^{2}k_{\bot}}{(2\pi)^{3}}\left|E_{\mathbf{k},\Vert,0}\right|^{2}(\mathbf{k}_{\bot})\frac{1}{v_{Te}}\]


Les coefficients de diffusion quasilineaire pour l'onde hybride deviennent
alors \begin{eqnarray*}
\widetilde{D}^{\text{LH}(0)}(p,\xi_{0}) & \equiv & \sigma\left\{ \frac{\sigma\xi}{\xi_{0}}D^{\text{LH}}(p,\xi)\right\} \\
 & = & \sigma\left\{ \frac{\sigma\xi}{\xi_{0}}\frac{\xi^{2}}{1-\xi^{2}}\frac{v_{Te}}{v\left|\xi\right|}H(v\xi-v_{\min})H(v_{\max}-v\xi)2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\right\} \\
 & = & \frac{\xi_{0}^{2}}{1-\xi_{0}^{2}}\frac{v_{Te}}{v\xi_{0}}\left\{ \frac{\sigma}{\Psi}\frac{\xi^{2}}{\xi_{0}^{2}}H(v\xi-v_{\min})H(v_{\max}-v\xi)2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\right\} \\
 & = & \frac{\xi_{0}^{2}}{1-\xi_{0}^{2}}\frac{v_{Te}}{v\xi_{0}}\frac{1}{\lambda\left(\xi_{0}\right)}\left[\frac{1}{2}\sum_{\sigma=\pm1}\right]_{T}\int_{-\theta_{c}}^{\theta_{c}}\frac{d\theta}{2\pi}\frac{\sigma\xi}{\xi_{0}\Psi}H(v\xi-v_{\min})H(v_{\max}-v\xi)2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\end{eqnarray*}
 et finalement \begin{eqnarray*}
\widetilde{D}^{\text{LH}(0)}(p,\xi_{0}) & \equiv & \sigma\left\{ \frac{\sigma\xi}{\xi_{0}}D^{\text{LH}}(p,\xi)\right\} \\
 & = & \frac{\xi_{0}^{2}}{1-\xi_{0}^{2}}\frac{v_{Te}}{v\xi_{0}}\frac{1}{\lambda\left(\xi_{0}\right)}\left[\frac{1}{2}\sum_{\sigma=\pm1}\right]_{T}\frac{\sigma\xi_{b}}{\xi_{0}\Psi_{b}}H(|\theta_{b}|<\theta_{c}\left(\xi_{0}\right))H(v\xi_{b}\left(\xi_{0}\right)-v_{\min,b})H(v_{\max,b}-v\xi_{b}\left(\xi_{0}\right))D_{0,b}^{\text{LH}}\end{eqnarray*}
 avec \[
D_{0,b}^{\text{LH}}=\lim_{V\rightarrow\infty}\frac{\pi q_{e}^{2}\Delta\theta_{b}}{V}\int\frac{d^{2}k_{\bot}}{(2\pi)^{3}}\left|E_{\mathbf{k},\Vert,0,b}\right|^{2}(\mathbf{k}_{\bot})\frac{1}{v_{Te}}\]


De meme

\begin{eqnarray*}
\widetilde{D}_{F}^{\text{LH}(0)}(p,\xi_{0}) & \equiv & \sigma\left\{ \left(\Psi(\theta)-1\right)\frac{\sigma\xi_{0}}{\xi}D^{\text{LH}}(p,\xi)\right\} \\
 & = & \sigma\left\{ \left(\Psi(\theta)-1\right)\frac{\sigma\xi_{0}}{\xi}\frac{\xi^{2}}{1-\xi^{2}}\frac{v_{Te}}{v\left|\xi\right|}H(v\xi-v_{\min})H(v_{\max}-v\xi)2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\right\} \\
 & = & \frac{\xi_{0}^{2}}{1-\xi_{0}^{2}}\frac{v_{Te}}{v\xi_{0}}\left\{ \left(\Psi-1\right)\frac{\sigma}{\Psi}H(v\xi-v_{\min})H(v_{\max}-v\xi)2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\right\} \\
 & = & \frac{\xi_{0}^{2}}{1-\xi_{0}^{2}}\frac{v_{Te}}{v\xi_{0}}\frac{1}{\lambda\left(\xi_{0}\right)}\left[\frac{1}{2}\sum_{\sigma=\pm1}\right]_{T}\int_{-\theta_{c}}^{\theta_{c}}\frac{d\theta}{2\pi}\left(\Psi-1\right)\frac{\sigma\xi_{0}}{\xi\Psi}H(v\xi-v_{\min})H(v_{\max}-v\xi)2\pi\delta(\theta-\theta_{b})D_{0}^{\text{LH}}\end{eqnarray*}
 et finalement \begin{eqnarray*}
\widetilde{D}_{F}^{\text{LH}(0)}(p,\xi_{0}) & \equiv & \sigma\left\{ \left(\Psi(\theta)-1\right)\frac{\sigma\xi_{0}}{\xi}D^{\text{LH}}(p,\xi)\right\} \\
 & = & \frac{\xi_{0}^{2}}{1-\xi_{0}^{2}}\frac{v_{Te}}{v\xi_{0}}\frac{1}{\lambda\left(\xi_{0}\right)}\left[\frac{1}{2}\sum_{\sigma=\pm1}\right]_{T}\frac{\sigma\xi_{0}\left(\Psi_{b}-1\right)}{\xi_{b}\Psi_{b}}H(|\theta_{b}|<\theta_{c}\left(\xi_{0}\right))H(v\xi_{b}\left(\xi_{0}\right)-v_{\min,b})H(v_{\max,b}-v\xi_{b}\left(\xi_{0}\right))D_{0,b}^{\text{LH}}\end{eqnarray*}
