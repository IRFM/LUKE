
\subsubsection{Collisions}

Since first order drift kinetic terms may be expressed in a conservative
form as for the zero order Fokker-Planck equation, the determination of the
matrix elements is therefore straightforward. One obtains
\begin{equation}
\widetilde{\overline{\overline{D}}}_{p}^{C\left( 0\right) }\rightarrow
\left\{
\begin{array}{l}
\widetilde{D}_{pp,l+1/2,i+1,j+1/2}^{C\left( 0\right) }=\left( \overline{%
\lambda }_{1,-1,0}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) A_{l+1/2,i+1}
\\
\widetilde{D}_{pp,l+1/2,i,j+1/2}^{C\left( 0\right) }=\left( \overline{%
\lambda }_{1,-1,0}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) A_{l+1/2,i}
\\
\widetilde{D}_{p\xi ,l+1/2,i+1,j+1/2}^{C\left( 0\right) }=0 \\
\widetilde{D}_{\xi p,l+1/2,i+1/2,j+1}^{C\left( 0\right) }=0 \\
\widetilde{D}_{p\xi ,l+1/2,i+1/2,j+1/2}^{C\left( 0\right) }=0 \\
\widetilde{D}_{\xi p,l+1/2,i+1/2,j+1/2}^{C\left( 0\right) }=0 \\
\widetilde{D}_{p\xi ,l+1/2,i,j+1/2}^{C\left( 0\right) }=0 \\
\widetilde{D}_{\xi p,l+1/2,i+1/2,j}^{C\left( 0\right) }=0 \\
\widetilde{D}_{\xi \xi ,l+1/2,i+1/2,j+1}^{C\left( 0\right) }=\left(
\overline{\lambda }_{3,-2,0}^{l+1/2,j+1}/\lambda ^{l+1/2,j+1}\right)
B_{t,l+1/2,i+1/2} \\
\widetilde{D}_{\xi \xi ,l+1/2,i+1/2,j}^{C\left( 0\right) }=\left( \overline{%
\lambda }_{3,-2,0}^{l+1/2,j}/\lambda ^{l+1/2,j}\right) B_{t,l+1/2,i+1/2}
\end{array}
\right.
\end{equation}
and
\begin{equation}
\widetilde{\overline{\overline{F}}}_{p}^{C\left( 0\right) }\rightarrow
\left\{
\begin{array}{l}
\widetilde{F}_{p,l+1/2,i+1,j+1/2}^{C\left( 0\right) }=-\left( \overline{%
\lambda }_{1,-1,0}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) F_{l+1/2,i+1}
\\
\widetilde{F}_{p,l+1/2,i,j+1/2}^{C\left( 0\right) }=-\left( \overline{%
\lambda }_{1,-1,0}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) F_{l+1/2,i}
\\
\widetilde{F}_{\xi ,l+1/2,i+1/2,j+1}^{C\left( 0\right) }=\frac{\sqrt{1-\xi
_{0,j+1}^{2}}}{p_{i+1/2}\xi _{0,j+1}^{3}}\left( \frac{\overline{\lambda }%
_{1,-1,0}^{l+1/2,j+1}-\overline{\lambda }_{1,-2,0}^{l+1/2,j+1}}{\lambda
^{l+1/2,j+1}}\right) B_{t,l+1/2,i+1/2} \\
\widetilde{F}_{\xi ,l+1/2,i+1/2,j}^{C\left( 0\right) }=\frac{\sqrt{1-\xi
_{0,j}^{2}}}{p_{i+1/2}\xi _{0,j}^{3}}\left( \frac{\overline{\lambda }%
_{1,-1,0}^{l+1/2,j}-\overline{\lambda }_{1,-2,0}^{l+1/2,j}}{\lambda
^{l+1/2,j}}\right) B_{t,l+1/2,i+1/2}
\end{array}
\right.
\end{equation}
where coefficients of the collision operator $A,F$ and $B_{t}$ are the same
as defined for the zero order bounce averaged Fokker-Planck equation, in
Sec. \ref{fpdiscretephysproc}.

Concerning the first order Legendre correction for
electron-electron collisions that ensures momentum conservation,
one must calculate
\[
\left\{
C\left( f_{M},\widetilde{f}\right) \right\} =-\frac{3}{2}\frac{\xi _{0}}{%
\lambda }H\left( \left| \xi _{0}\right| -\xi _{0T}\right)
\mathcal{I}\left( f_{M},\widetilde{f}_{0}^{\left( 0\right) \left(
m=1\right) }\right)
\]
on the distribution function grid, where

\begin{equation}
\widetilde{f}_{0}^{\left( 0\right) \left( m=1\right) }\left( \psi ,p,\xi
_{0}\right) =\int_{-1}^{+1}\xi _{0}\widetilde{f}_{0}^{\left( 0\right)
}\left( \psi ,p,\xi _{0}\right) \overline{\lambda }_{2,0,0}d\xi _{0}
\end{equation}
as shwon in Sec. \ref{dkecollisions}.

Here, the collision integral $\mathcal{I}\left( f_{0},\widetilde{f}%
_{0}^{\left( 0\right) \left( m=1\right) }\right) $ has a similar form as for
the zero order Fokker-Planck equation except that $f_{0}^{\left( 0\right)
\left( m=1\right) }$ is just replaced by $\widetilde{f}_{0}^{\left( 0\right)
\left( m=1\right) }$ in all corresponding terms. Therefore, by definition,
at the iteration number $\left( k\right) ,$%
\begin{align}
\left\{ C\left( f_{M},\widetilde{f}\right) \right\}
_{l+1/2,i+1/2,j+1/2}^{\left( k\right) }&=-\frac{3}{2}\frac{\xi _{0,j+1/2}}{%
\lambda ^{l+1/2,j+1/2}}H\left( \left| \xi _{0,j+1/2}\right| -\xi
_{0T,l+1/2}\right)\nonumber\\&\times \mathcal{I}\left(
f_{M,},\widetilde{f}_{0}^{\left( 0\right) \left( m=1\right)
}\right) _{l+1/2,i+1/2,j+1/2}^{\left( k\right) }
\end{align}
where
\begin{equation}
\widetilde{f}_{0,l+1/2,i+1/2,j+1/2}^{\left( 0\right) \left( m=1\right)
\left( k\right) }=\sum_{j^{\prime }=0}^{n_{\xi _{0}-1}}\xi _{0,j^{\prime
}+1/2}\widetilde{f}_{0,l+1/2,i+1/2,j^{\prime }+1/2}^{\left( 0\right) \left(
k\right) }\overline{\lambda }_{2,0,0}^{l+1/2,j^{\prime }+1/2}\Delta \xi
_{0,j^{\prime }+1/2}
\end{equation}

\subsubsection{Ohmic electric field}

According to the bounce averaged expression,
\begin{equation}
\widetilde{\overline{\overline{D}}}_{p}^{E\left( 0\right) }\rightarrow
\left\{
\begin{array}{l}
\widetilde{D}_{pp,l+1/2,i+1,j+1/2}^{E\left( 0\right) }=0 \\
\widetilde{D}_{pp,l+1/2,i,j+1/2}^{E\left( 0\right) }=0 \\
\widetilde{D}_{p\xi ,l+1/2,i+1,j+1/2}^{E\left( 0\right) }=0 \\
\widetilde{D}_{\xi p,l+1/2,i+1/2,j+1}^{E\left( 0\right) }=0 \\
\widetilde{D}_{p\xi ,l+1/2,i+1/2,j+1/2}^{E\left( 0\right) }=0 \\
\widetilde{D}_{\xi p,l+1/2,i+1/2,j+1/2}^{E\left( 0\right) }=0 \\
\widetilde{D}_{p\xi ,l+1/2,i,j+1/2}^{E\left( 0\right) }=0 \\
\widetilde{D}_{\xi p,l+1/2,i+1/2,j}^{E\left( 0\right) }=0 \\
\widetilde{D}_{\xi \xi ,l+1/2,i+1/2,j+1}^{E\left( 0\right) }=0 \\
\widetilde{D}_{\xi \xi ,l+1/2,i+1/2,j}^{E\left( 0\right) }=0
\end{array}
\right.
\end{equation}
and
\begin{equation}
\widetilde{\overline{\overline{F}}}_{p}^{E\left( 0\right) }\rightarrow
\left\{
\begin{array}{l}
\widetilde{F}_{p,l+1/2,i+1,j+1/2}^{E\left( 0\right) }=\left( \overline{%
\lambda }_{2,-2,2}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) \xi _{0,j+1/2}%
\overline{E}_{\Vert 0,l+1/2} \\
\widetilde{F}_{p,l+1/2,i,j+1/2}^{E\left( 0\right) }=\left( \overline{\lambda
}_{2,-2,2}^{l+1/2,j+1/2}/\lambda ^{l+1/2,j+1/2}\right) \xi _{0,j+1/2}%
\overline{E}_{\Vert 0,l+1/2} \\
\widetilde{F}_{\xi ,l+1/2,i+1/2,j+1}^{E\left( 0\right) }=-\left( \overline{%
\lambda }_{2,-2,2}^{l+1/2,j+1}/\lambda ^{l+1/2,j+1}\right) \sqrt{1-\xi
_{0,j+1}^{2}}\overline{E}_{\Vert 0,l+1/2} \\
\widetilde{F}_{\xi ,l+1/2,i+1/2,j}^{E\left( 0\right) }=-\left( \overline{%
\lambda }_{2,-2,2}^{l+1/2,j}/\lambda ^{l+1/2,j}\right) \sqrt{1-\xi _{0,j}^{2}%
}\overline{E}_{\Vert 0,l+1/2}
\end{array}
\right.
\end{equation}
where $\overline{E}_{\Vert 0,l+1/2}$ is the parallel component of the Ohmic
electric field along the magnetic field line direction normalized to the
Dreicer field taken at the poloidal position where the magnetic field $%
\mathbf{B}$ is minimum, as explained in Sec.\ref{Efield}.
