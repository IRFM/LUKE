In the Fokker-Planck equation, the fluxes are expressed as (). 
\[
{\bf S}_{{\bf p}}=-{\Bbb D}_{p}\cdot \nabla _{p,\xi }f_{0}+{\bf F}_{p}f_{0}
\]
with
\begin{eqnarray*}
S_{p} &=&-D_{pp}\frac{\partial f_{0}}{\partial p}+D_{p\xi }\frac{\sqrt{1-\xi
^{2}}}{p}\frac{\partial f_{0}}{\partial \xi }+F_{p}f_{0} \\
S_{\xi } &=&-D_{\xi p}\frac{\partial f_{0}}{\partial p}+D_{\xi \xi }\frac{%
\sqrt{1-\xi ^{2}}}{p}\frac{\partial f_{0}}{\partial \xi }+F_{\xi }f_{0}
\end{eqnarray*}
and we need to evaluate the bounce-averaged fluxes (\ref{sp0})
\begin{eqnarray}
S_{p}^{\left( 0\right) } &=&\left\{ S_{p}\right\}  \\
S_{\xi _{0}}^{\left( 0\right) } &=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{%
\Psi }\xi _{0}}S_{\xi }\right\} 
\end{eqnarray}

Because $f_{0}$ is constant along a field line, we have $f_{0}\left( p,\xi
\right) =f_{0}^{(0)}\left( p,\xi _{0}\right) $which is independent of $%
\theta $ and $\sigma $. Using the following identities
\begin{eqnarray*}
\left\{ -D_{pp}\frac{\partial f_{0}}{\partial p}\right\}  &=&-\left\{
D_{pp}\right\} \frac{\partial f_{0}^{(0)}}{\partial p} \\
\left\{ D_{p\xi }\frac{\sqrt{1-\xi ^{2}}}{p}\frac{\partial f_{0}}{\partial %
\xi }\right\}  &=&\frac{\sqrt{1-\xi _{0}^{2}}}{p}\left\{ \frac{\sigma \xi }{%
\sqrt{\Psi }\xi _{0}}D_{p\xi }\right\} \frac{\partial f_{0}^{(0)}}{\sigma 
\partial \xi _{0}} \\
\left\{ F_{p}f_{0}\right\}  &=&\left\{ F_{p}\right\} f_{0}^{(0)} \\
-\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}D_{\xi p}\frac{%
\partial f_{0}}{\partial p}\right\}  &=&-\sigma \left\{ \frac{\sigma \xi }{%
\sqrt{\Psi }\xi _{0}}D_{\xi p}\right\} \frac{\partial f_{0}^{(0)}}{\partial p%
} \\
-\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}D_{\xi \xi }\frac{%
\sqrt{1-\xi ^{2}}}{p}\frac{\partial f_{0}}{\partial \xi }\right\} 
&=&-\sigma \frac{\sqrt{1-\xi _{0}^{2}}}{p}\left\{ \frac{\xi ^{2}}{\Psi \xi
_{0}^{2}}D_{\xi \xi }\right\} \frac{\partial f_{0}^{(0)}}{\sigma \partial %
\xi _{0}} \\
\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}F_{\xi }f_{0}\right\}
&=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}F_{\xi }\right\}
f_{0}^{(0)}
\end{eqnarray*}
we can rewrite
\[
{\bf S}_{{\bf p}}^{(0)}=-{\Bbb D}_{{\bf p}}^{(0)}\cdot \nabla _{p,\xi
_{0}}f_{0}^{(0)}+{\bf F}_{{\bf p}}^{(0)}f_{0}^{(0)}
\]
by defining the diffusin components
\begin{eqnarray*}
D_{pp}^{(0)} &=&\left\{ D_{pp}\right\}  \\
D_{p\xi }^{(0)} &=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}%
D_{p\xi }\right\}  \\
D_{\xi p}^{(0)} &=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}%
D_{\xi p}\right\}  \\
D_{\xi \xi }^{(0)} &=&\left\{ \frac{\xi ^{2}}{\Psi \xi _{0}^{2}}D_{\xi \xi
}\right\} 
\end{eqnarray*}
and the convection components
\begin{eqnarray*}
F_{p}^{(0)} &=&\left\{ F_{p}\right\}  \\
F_{\xi }^{(0)} &=&\sigma \left\{ \frac{\sigma \xi }{\sqrt{\Psi }\xi _{0}}%
F_{\xi }\right\} 
\end{eqnarray*}
where the gradient in the $\left( p,\xi _{0}\right) $ space is
\begin{equation}
\nabla _{p,\xi _{0}}=\left( 
\begin{array}{l}
\nabla _{p} \\ 
\nabla _{\xi _{0}}
\end{array}
\right) 
\end{equation}
with
\begin{eqnarray}
\nabla _{p} &=&\frac{\partial }{\partial p} \\
\nabla _{\xi _{0}} &=&\frac{-\sqrt{1-\xi _{0}^{2}}}{p}\frac{\partial }{%
\partial \xi _{0}}
\end{eqnarray}

The bounce averaged flux is therefore
\begin{equation}
{\bf S}_{{\bf p}}^{(0)}=\left( 
\begin{array}{l}
S_{p}^{(0)} \\ 
S_{\xi _{0}}^{(0)}
\end{array}
\right) 
\end{equation}
with
\begin{eqnarray}
S_{p}^{(0)} &=&-D_{pp}^{(0)}\frac{\partial f_{0}^{(0)}}{\partial p}+\frac{%
\sqrt{1-\xi _{0}^{2}}}{p}D_{p\xi }^{(0)}\frac{\partial f_{0}^{(0)}}{\partial %
\xi _{0}}+F_{p}^{(0)}f_{0}^{(0)} \\
S_{\xi _{0}}^{(0)} &=&-D_{\xi p}^{(0)}\frac{\partial f_{0}^{(0)}}{\partial p}%
+\frac{\sqrt{1-\xi _{0}^{2}}}{p}D_{\xi \xi }^{(0)}\frac{\partial f_{0}^{(0)}%
}{\partial \xi _{0}}+F_{\xi }^{(0)}f_{0}^{(0)}
\end{eqnarray}
The momentum space dynamics operator of the Bounce-averaged Fokker-Planck
equation (\ref{kinbav2}) therefore expands as
\begin{eqnarray}
\left\{ J_{{\bf p}}\left. \frac{\partial f_{0}}{\partial t}\right| _{{\bf p}%
}\right\}  &=&\frac{\partial }{\partial p}\left( -p^{2}D_{pp}^{(0)}\frac{%
\partial f_{0}^{(0)}}{\partial p}+p^{2}F_{p}^{(0)}f_{0}^{(0)}\right) 
\label{kinbav2} \\
&&+\sqrt{1-\xi _{0}^{2}}\frac{\partial }{\partial p}\left( pD_{p\xi
}^{(0)}\right) \frac{\partial f_{0}^{(0)}}{\partial \xi _{0}}+\sqrt{1-\xi
_{0}^{2}}pD_{p\xi }^{(0)}\frac{\partial ^{2}f_{0}^{(0)}}{\partial p\partial %
\xi _{0}} \\
&&-\frac{p}{\lambda \left( \xi _{0}\right) }\frac{\partial }{\partial \xi
_{0}}\left( \lambda \left( \xi _{0}\right) D_{\xi \xi }^{(0)}\frac{1-\xi
_{0}^{2}}{p}\frac{\partial f_{0}^{(0)}}{\partial \xi _{0}}+\lambda \left(
\xi _{0}\right) \sqrt{1-\xi _{0}^{2}}F_{\xi }f_{0}^{(0)}\right)  \\
&&+\frac{p}{\lambda \left( \xi _{0}\right) }\frac{\partial }{\partial \xi
_{0}}\left( \lambda \left( \xi _{0}\right) \sqrt{1-\xi _{0}^{2}}D_{\xi
p}^{(0)}\right) \frac{\partial f_{0}^{(0)}}{\partial p}+p\sqrt{1-\xi _{0}^{2}%
}D_{\xi p}^{(0)}\frac{\partial ^{2}f_{0}^{(0)}}{\partial \xi _{0}\partial p}
\end{eqnarray}
