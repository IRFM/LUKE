The bounce averaged Fokker-Planck equation is is given in the
conservative form by relation (\ref{kinbav2}), with the
bounce-averaged fluxes (\ref{sp0}) and (\ref{smu0})
\begin{align}
S_{p}^{\left(  0\right)  }  & =\left\{  S_{p}\right\}  \\
S_{\xi_{0}}^{\left(  0\right)  }  & =\sigma\left\{  \frac{\sigma\xi}%
{\sqrt{\Psi}\xi_{0}}S_{\xi}\right\}
\end{align}

Because $f_{0}$ is constant along a magnetic field line, we have $f_{0}\left(
p,\xi\right)  =f_{0}^{(0)}\left(  p,\xi_{0}\right)  $ which is independent of
$\theta$ and $\sigma$. Using the following identities
\begin{align}
-\left\{  D_{pp}\frac{\partial f_{0}}{\partial p}\right\}    & =-\left\{
D_{pp}\right\}  \frac{\partial f_{0}^{(0)}}{\partial p}\\
\left\{  D_{p\xi}\frac{\sqrt{1-\xi^{2}}}{p}\frac{\partial f_{0}}{\partial\xi
}\right\}    & =\frac{\sqrt{1-\xi_{0}^{2}}}{p}\left\{  \frac{\sigma\xi}%
{\sqrt{\Psi}\xi_{0}}D_{p\xi}\right\}  \frac{\partial f_{0}^{(0)}}%
{\sigma\partial\xi_{0}}\\
\left\{  F_{p}f_{0}\right\}    & =\left\{  F_{p}\right\}  f_{0}^{(0)}\\
-\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}D_{\xi p}\frac{\partial
f_{0}}{\partial p}\right\}    & =-\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi
}\xi_{0}}D_{\xi p}\right\}  \frac{\partial f_{0}^{(0)}}{\partial p}\\
\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}D_{\xi\xi}\frac{\sqrt
{1-\xi^{2}}}{p}\frac{\partial f_{0}}{\partial\xi}\right\}    & =\frac
{\sqrt{1-\xi_{0}^{2}}}{p}\sigma\left\{  \frac{\xi^{2}}{\Psi\xi_{0}^{2}}%
D_{\xi\xi}\right\}  \frac{\partial f_{0}^{(0)}}{\sigma\partial\xi_{0}}\\
\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}F_{\xi}f_{0}\right\}    &
=\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}F_{\xi}\right\}
f_{0}^{(0)}%
\end{align}
we can rewrite
\begin{equation}
\mathbf{S}_{\mathbf{p}}^{(0)}=-\mathbb{{D}_{\mathbf{p}}^{(0)}\cdot
\nabla_{p,\xi_{0}}f_{0}^{(0)}+\mathbf{F}_{\mathbf{p}}^{(0)}f_{0}^{(0)}}%
\end{equation}
where the bounce averaged flux is decomposed into
\begin{equation}
\mathbf{S}_{\mathbf{p}}^{(0)}=\left(
\begin{array}
[c]{l}%
S_{p}^{(0)}\\
S_{\xi_{0}}^{(0)}%
\end{array}
\right)
\end{equation}
with
\begin{align}
S_{p}^{(0)}  & =-D_{pp}^{(0)}\frac{\partial f_{0}^{(0)}}{\partial p}%
+\frac{\sqrt{1-\xi_{0}^{2}}}{p}D_{p\xi}^{(0)}\frac{\partial f_{0}^{(0)}%
}{\partial\xi_{0}}+F_{p}^{(0)}f_{0}^{(0)}\label{sp0dev}\\
S_{\xi_{0}}^{(0)}  & =-D_{\xi p}^{(0)}\frac{\partial f_{0}^{(0)}}{\partial
p}+\frac{\sqrt{1-\xi_{0}^{2}}}{p}D_{\xi\xi}^{(0)}\frac{\partial f_{0}^{(0)}%
}{\partial\xi_{0}}+F_{\xi}^{(0)}f_{0}^{(0)}%
\end{align}
by defining the diffusion components
\begin{align}
D_{pp}^{(0)}  & =\left\{  D_{pp}\right\}  \label{diffbavpp}\\
D_{p\xi}^{(0)}  & =\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}D_{p\xi
}\right\}  \label{diffbavpm}\\
D_{\xi p}^{(0)}  & =\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}D_{\xi
p}\right\}  \label{diffbavmp}\\
D_{\xi\xi}^{(0)}  & =\left\{  \frac{\xi^{2}}{\Psi\xi_{0}^{2}}D_{\xi\xi
}\right\}  \label{diffbavmm}%
\end{align}
and the convection components
\begin{align}
F_{p}^{(0)}  & =\left\{  F_{p}\right\}  \label{convbavp}\\
F_{\xi}^{(0)}  & =\sigma\left\{  \frac{\sigma\xi}{\sqrt{\Psi}\xi_{0}}F_{\xi
}\right\}  \label{convbavm}%
\end{align}
where the gradient vector in the reduced $\left(  p,\xi_{0}\right)  $ momentum
space is
\begin{equation}
\nabla_{p,\xi_{0}}=\left(
\begin{array}
[c]{l}%
\nabla_{p}\\
\nabla_{\xi_{0}}%
\end{array}
\right)
\end{equation}
with
\begin{align}
\nabla_{p}  & =\frac{\partial}{\partial p}\\
\nabla_{\xi_{0}}  & =\frac{-\sqrt{1-\xi_{0}^{2}}}{p}\frac{\partial}%
{\partial\xi_{0}}%
\end{align}
